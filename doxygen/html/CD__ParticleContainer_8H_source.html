193 void
194 define(const Vector<DisjointBoxLayout>& a_grids,
195 const Vector<ProblemDomain>& a_domains,
196 const Vector<Real>& a_dx,
197 const Vector<int>& a_refRat,
198 const RealVect& a_probLo,
199 const int a_minBlockSize,
200 const Vector<RefCountedPtr<LevelTiles>>& a_levelTiles,
201 const int a_finestLevel,
202 const std::string& a_realm,
203 const Vector<RefCountedPtr<LevelData<BaseFab<bool>>>>* a_validCells)
204 {
205 CH_TIME("ParticleContainer::define");
206
207 CH_assert(a_grids.size() > a_finestLevel);
208 CH_assert(a_domains.size() > a_finestLevel);
209 CH_assert(a_dx.size() > a_finestLevel);
210 CH_assert(a_refRat.size() > a_finestLevel);
211 CH_assert(a_levelTiles.size() > a_finestLevel);
212
213 m_grids = a_grids;
214 m_domains = a_domains;
215 m_refRat = a_refRat;
216 m_probLo = a_probLo;
217 m_minBlockSize = a_minBlockSize;
218 m_finestLevel = a_finestLevel;
219 m_realm = a_realm;
220 m_validCells = a_validCells;
221 m_numOutcastLocal = 0;
222
223 m_dx.resize(a_finestLevel + 1);
224 for (int lvl = 0; lvl <= a_finestLevel; lvl++) {
225 m_dx[lvl] = a_dx[lvl] * RealVect::Unit;
226 }
227
228 for (int dir = 0; dir < SpaceDim; dir++) {
229 m_probHi[dir] = m_probLo[dir] + a_domains[0].domainBox().size(dir) * a_dx[0];
230 }
231
232 // Alias the Realm's per-level tile->box maps (the single source of truth); never build our own.
233 m_levelTiles = a_levelTiles;
234
235 m_particles.resize(a_finestLevel + 1);
236 m_maskParticles.resize(a_finestLevel + 1);
237 for (int lvl = 0; lvl <= a_finestLevel; lvl++) {
238 m_particles[lvl] = RefCountedPtr<LevelParticles>(new LevelParticles(a_grids[lvl]));
239 m_maskParticles[lvl] = RefCountedPtr<LevelParticles>(new LevelParticles(a_grids[lvl]));
240 }
241
242 // Buffer particles live on the grown grids (boxes grown by the refinement factor on finer levels).
243 this->setupGrownGrids();
244 m_bufferParticles.resize(a_finestLevel + 1);
245 for (int lvl = 0; lvl <= a_finestLevel; lvl++) {
246 m_bufferParticles[lvl] = RefCountedPtr<LevelParticles>(new LevelParticles(m_grownGrids[lvl]));
247 }
248
249 m_isOrganizedByCell = false;
250 m_isDefined = true;
251 }

513 unsigned long long
514 getNumberOfValidParticlesLocal() const
515 {
516 CH_TIME("ParticleContainer::getNumberOfValidParticlesLocal");
517
518 CH_assert(m_isDefined);
519
520 unsigned long long n = 0;
521 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
522 const LevelParticles& level = *m_particles[lvl];
523 const DisjointBoxLayout& dbl = m_grids[lvl];
524 const DataIterator& dit = dbl.dataIterator();
525 const int nbox = dit.size();
526
527#pragma omp parallel for schedule(runtime) reduction(+ : n)
528 for (int mybox = 0; mybox < nbox; mybox++) {
529 const DataIndex& din = dit[mybox];
530 const Leaf& leaf = level[din];
531
532 for (std::size_t i = 0; i < leaf.size(); i++) {
533 if (!leaf.isGhost(i)) {
534 n++;
535 }
536 }
537 }
538 }
539
540 return n;
541 }

810 transferParticles(AMRParticlesSoA<P, Traits>& a_source)
811 {
812 CH_TIME("ParticleContainer::transferParticles");
813
814 CH_assert(m_isDefined);
815
816 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
817 LevelParticles& src = *a_source[lvl];
818 LevelParticles& dst = *m_particles[lvl];
819
820 const DisjointBoxLayout& dbl = m_grids[lvl];
821 const DataIterator& dit = dbl.dataIterator();
822
823 const int nbox = dit.size();
824
825#pragma omp parallel for schedule(runtime)
826 for (int mybox = 0; mybox < nbox; mybox++) {
827 const DataIndex& din = dit[mybox];
828
829 dst[din].catenate(src[din]); // bulk move (id/rank preserved); empty dst -> O(1) swap
830 }
831 }

1081 inline IntVect
1082 cellIndex(const int a_lvl, const RealVect& a_pos) const
1083 {
1084 IntVect iv;
1085 for (int dir = 0; dir < SpaceDim; dir++) {
1086 iv[dir] = static_cast<int>(std::floor((a_pos[dir] - m_probLo[dir]) / m_dx[a_lvl][dir]));
1087 }
1088 return iv;
1089 }

1101 inline bool
1102 moveInsideDomain(RealVect& a_pos) const noexcept
1103 {
1104 const Real margin = 1.E-6 * m_dx[m_finestLevel][0];
1105
1106 bool moved = false;
1107
1108 for (int dir = 0; dir < SpaceDim; dir++) {
1109 if (a_pos[dir] < m_probLo[dir] && a_pos[dir] >= m_probLo[dir] - margin) {
1110 a_pos[dir] = m_probLo[dir] + margin;
1111 moved = true;
1112 }
1113 else if (a_pos[dir] >= m_probHi[dir] - margin && a_pos[dir] <= m_probHi[dir] + margin) {
1114 a_pos[dir] = m_probHi[dir] - margin;
1115 moved = true;
1116 }
1117 }
1118
1119 return moved;
1120 }
