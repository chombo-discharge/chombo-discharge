193 void
194 define(const Vector<DisjointBoxLayout>& a_grids,
195 const Vector<ProblemDomain>& a_domains,
196 const Vector<Real>& a_dx,
197 const Vector<int>& a_refRat,
198 const RealVect& a_probLo,
199 const int a_minBlockSize,
200 const Vector<RefCountedPtr<LevelTiles>>& a_levelTiles,
201 const int a_finestLevel,
202 const std::string& a_realm,
203 const Vector<RefCountedPtr<LevelData<BaseFab<bool>>>>* a_validCells)
204 {
205 CH_TIME("ParticleContainer::define");
206
207 CH_assert(a_grids.size() > a_finestLevel);
208 CH_assert(a_domains.size() > a_finestLevel);
209 CH_assert(a_dx.size() > a_finestLevel);
210 CH_assert(a_refRat.size() > a_finestLevel);
211 CH_assert(a_levelTiles.size() > a_finestLevel);
212
213 m_grids = a_grids;
214 m_domains = a_domains;
215 m_refRat = a_refRat;
216 m_probLo = a_probLo;
217 m_minBlockSize = a_minBlockSize;
218 m_finestLevel = a_finestLevel;
219 m_realm = a_realm;
220 m_validCells = a_validCells;
221 m_numOutcastLocal = 0;
222
223 m_dx.resize(a_finestLevel + 1);
224 for (int lvl = 0; lvl <= a_finestLevel; lvl++) {
225 m_dx[lvl] = a_dx[lvl] * RealVect::Unit;
226 }
227
228 for (int dir = 0; dir < SpaceDim; dir++) {
229 m_probHi[dir] = m_probLo[dir] + a_domains[0].domainBox().size(dir) * a_dx[0];
230 }
231
232 // Alias the Realm's per-level tile->box maps (the single source of truth); never build our own.
233 m_levelTiles = a_levelTiles;
234
235 m_particles.resize(a_finestLevel + 1);
236 m_maskParticles.resize(a_finestLevel + 1);
237 m_maskOrigins.resize(a_finestLevel + 1);
238 for (int lvl = 0; lvl <= a_finestLevel; lvl++) {
239 m_particles[lvl] = RefCountedPtr<LevelParticles>(new LevelParticles(a_grids[lvl]));
240 m_maskParticles[lvl] = RefCountedPtr<LevelParticles>(new LevelParticles(a_grids[lvl]));
241 m_maskOrigins[lvl] = RefCountedPtr<LayoutData<std::vector<std::size_t>>>(
242 new LayoutData<std::vector<std::size_t>>(a_grids[lvl]));
243 }
244
245 // Buffer particles live on the grown grids (boxes grown by the refinement factor on finer levels).
246 this->setupGrownGrids();
247 m_bufferParticles.resize(a_finestLevel + 1);
248 for (int lvl = 0; lvl <= a_finestLevel; lvl++) {
249 m_bufferParticles[lvl] = RefCountedPtr<LevelParticles>(new LevelParticles(m_grownGrids[lvl]));
250 }
251
252 m_isOrganizedByCell = false;
253 m_isDefined = true;
254 }

486 addParticlesDestructive(ParticleSoA<P, Traits>& a_particles)
487 {
488 CH_TIME("ParticleContainer::addParticlesDestructive");
489
490 CH_assert(m_isDefined);
491
492 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
493 DataIterator dit = m_grids[lvl].dataIterator();
494 if (dit.ok()) {
495 (*m_particles[lvl])[dit()].catenate(a_particles); // move the buffer onto a local leaf

516 unsigned long long
517 getNumberOfValidParticlesLocal() const
518 {
519 CH_TIME("ParticleContainer::getNumberOfValidParticlesLocal");
520
521 CH_assert(m_isDefined);
522
523 unsigned long long n = 0;
524 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
525 const LevelParticles& level = *m_particles[lvl];
526 const DisjointBoxLayout& dbl = m_grids[lvl];
527 const DataIterator& dit = dbl.dataIterator();
528 const int nbox = dit.size();
529
530#pragma omp parallel for schedule(runtime) reduction(+ : n)
531 for (int mybox = 0; mybox < nbox; mybox++) {
532 const DataIndex& din = dit[mybox];
533 const Leaf& leaf = level[din];
534
535 for (std::size_t i = 0; i < leaf.size(); i++) {
536 if (!leaf.isGhost(i)) {
537 n++;
538 }
539 }
540 }
541 }
542
543 return n;
544 }

861 void
862 transferParticles(AMRParticlesSoA<P, Traits>& a_source)
863 {
864 CH_TIME("ParticleContainer::transferParticles");
865
866 CH_assert(m_isDefined);
867
868 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
869 LevelParticles& src = *a_source[lvl];
870 LevelParticles& dst = *m_particles[lvl];
871
872 const DisjointBoxLayout& dbl = m_grids[lvl];
873 const DataIterator& dit = dbl.dataIterator();
874
875 const int nbox = dit.size();
876
877#pragma omp parallel for schedule(runtime)
878 for (int mybox = 0; mybox < nbox; mybox++) {
879 const DataIndex& din = dit[mybox];
880
881 dst[din].catenate(src[din]); // bulk move (id/rank preserved); empty dst -> O(1) swap
882 }
883 }

1154 inline IntVect
1155 cellIndex(const int a_lvl, const RealVect& a_pos) const
1156 {
1157 IntVect iv;
1158 for (int dir = 0; dir < SpaceDim; dir++) {
1159 iv[dir] = static_cast<int>(std::floor((a_pos[dir] - m_probLo[dir]) / m_dx[a_lvl][dir]));
1160 }
1161 return iv;
1162 }

1174 inline bool
1175 moveInsideDomain(RealVect& a_pos) const noexcept
1176 {
1177 const Real margin = 1.E-6 * m_dx[m_finestLevel][0];
1178
1179 bool moved = false;
1180
1181 for (int dir = 0; dir < SpaceDim; dir++) {
1182 if (a_pos[dir] < m_probLo[dir] && a_pos[dir] >= m_probLo[dir] - margin) {
1183 a_pos[dir] = m_probLo[dir] + margin;
1184 moved = true;
1185 }
1186 else if (a_pos[dir] >= m_probHi[dir] - margin && a_pos[dir] <= m_probHi[dir] + margin) {
1187 a_pos[dir] = m_probHi[dir] - margin;
1188 moved = true;
1189 }
1190 }
1191
1192 return moved;
1193 }
