47template <typename P, typename Traits>
48inline typename ParticleContainer<P, Traits>::MoverPool
49ParticleContainer<P, Traits>::gatherToPool(AMRParticlesSoA<P, Traits>& a_source,
50 const Vector<DisjointBoxLayout>& a_sourceGrids)
51{
52 CH_TIME("ParticleContainer::gatherToPool");
53
54 MoverPool pool(numProc());
55
56 // Bin every particle of one leaf into a destination pool, returning the number of off-domain
57 // (outcast) particles. Shared by the serial and OpenMP paths so the binning logic lives once.
58 // findDestination() is const and reads only the (immutable-during-gather) tile maps, so it is
59 // safe to call concurrently; each thread bins into its own thread-local pool.
60 auto binLeaf = [this](MoverPool& a_target, const Leaf& a_leaf) -> unsigned long long {
61 unsigned long long outcast = 0;
62
63 for (std::size_t i = 0; i < a_leaf.size(); i++) {
64 RealVect pos = a_leaf.position(i);
65 auto dst = this->findDestination(pos);
66
67 if (!dst.valid && this->moveInsideDomain(pos)) {
68 dst = this->findDestination(pos);
69 }
70
71 if (!dst.valid) {
72 outcast++;
73 continue;
74 }
75
76 Leaf& bin = a_target[dst.rank][PoolKey(dst.level, dst.gridIndex)];
77
78 bin.append(pos, a_leaf.weight(i), a_leaf.gather(i));
79
80 const std::size_t k = bin.size() - 1;
81
82 bin.particleID(k) = a_leaf.particleID(i); // preserve id
83 bin.rankID(k) = dst.rank; // new owner
84 }
85
86 return outcast;
87 };
88
89 unsigned long long numOutcast = 0;
90
91 for (int lvl = 0; lvl < a_source.size(); lvl++) {
92 LevelParticles& level = *a_source[lvl];
93 const DataIterator& dit = a_sourceGrids[lvl].dataIterator();
94 const int nbox = dit.size();
95
96#ifdef _OPENMP
97#pragma omp parallel reduction(+ : numOutcast)
98 {
99 MoverPool threadPool(numProc()); // thread-local; no locking while binning
100
101#pragma omp for schedule(runtime)
102 for (int mybox = 0; mybox < nbox; mybox++) {
103 Leaf& leaf = level[dit[mybox]];
104
105 numOutcast += binLeaf(threadPool, leaf);
106
107 // The pool now holds this leaf's particles, and the source is discarded by every caller, so
108 // release the arena here rather than at the end. Otherwise the source and the pool are both
109 // fully live until the caller returns, which doubles the particle footprint at exactly the
110 // point in a regrid where it is already highest. Each thread owns its own boxes, so this is
111 // race-free.
112 leaf.clear();
113 leaf.shrinkToFit();
114 }
115
116 // Merge the thread-local pool into the shared pool. First writer to a (rank,key) slot gets the
117 // O(1) arena swap inside catenate(); later writers append.
118#pragma omp critical(ParticleContainer_gatherMerge)
119 {
120 for (int r = 0; r < numProc(); r++) {
121 for (auto& kv : threadPool[r]) {
122 pool[r][kv.first].catenate(kv.second);
123 }
124 }
125 }
126 }
127#else
128 for (int mybox = 0; mybox < nbox; mybox++) {
129 Leaf& leaf = level[dit[mybox]];
130
131 numOutcast += binLeaf(pool, leaf);
132
133 // See the OpenMP branch: drain as we go so the source and the pool are never both fully live.
134 leaf.clear();
135 leaf.shrinkToFit();
136 }
137#endif
138 }
139
140 m_numOutcastLocal = numOutcast;
141
142 return pool;
143}

145template <typename P, typename Traits>
146inline typename ParticleContainer<P, Traits>::MoverPool
147ParticleContainer<P, Traits>::gatherMoversToPool()
148{
149 CH_TIME("ParticleContainer::gatherMoversToPool");
150
151 CH_assert(m_validCells != nullptr);
152
153 MoverPool pool(numProc());
154 unsigned long long numOutcast = 0;
155
156 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
157 LevelParticles& level = *m_particles[lvl];
158 const LevelData<BaseFab<bool>>* maskLevel = (*m_validCells)[lvl].isNull() ? nullptr : &(*(*m_validCells)[lvl]);
159 const DataIterator& dit = m_grids[lvl].dataIterator();
160 const int nbox = dit.size();
161
162 // Partition one leaf in place: keep stayers (still inside their box AND in a cell flagged valid =
163 // not covered by a finer level), pool the movers and swap-pop them out. a_mask == nullptr forces
164 // every particle to be treated as a mover (no usable mask for this leaf). Returns outcast movers.
165 // findDestination()/cellIndex() are const reads, so this is safe to call from many threads as long
166 // as each leaf is touched by a single thread (the omp-for partitions boxes disjointly).
167 auto partitionLeaf = [this, lvl](MoverPool& a_target,
168 [[maybe_unused]] const DataIndex& a_din,
169 Leaf& a_leaf,
170 const BaseFab<bool>* a_mask) -> unsigned long long {
171 unsigned long long outcast = 0;
172
173 const Box maskBox = (a_mask != nullptr) ? a_mask->box() : Box();
174
175 std::size_t i = 0;
176
177 while (i < a_leaf.size()) {
178 RealVect pos = a_leaf.position(i);
179 if (a_mask != nullptr) {
180 const IntVect iv = this->cellIndex(lvl, pos);
181
182 if (maskBox.contains(iv) && (*a_mask)(iv, 0)) {
183 // Stayer: in its own box and not under a finer grid, so it cannot have changed owner.
184 // Debug self-check of that invariant -- the destination the gather-all path would compute
185 // must be exactly this (level, grid, rank). If this ever fires, the skip is unsound.
186#ifndef NDEBUG
187 const auto chk = this->findDestination(pos);
188 CH_assert(chk.valid && chk.level == lvl && chk.rank == procID() &&
189 m_levelTiles[lvl]->getMyGrids().at(chk.gridIndex) == a_din);
190#endif
191 i++;
192
193 continue;
194 }
195 }
196
197 auto dst = this->findDestination(pos); // mover: route it
198
199 if (!dst.valid && this->moveInsideDomain(pos)) {
200 dst = this->findDestination(pos);
201 }
202
203 if (dst.valid) {
204 Leaf& bin = a_target[dst.rank][PoolKey(dst.level, dst.gridIndex)];
205
206 bin.append(pos, a_leaf.weight(i), a_leaf.gather(i));
207
208 const std::size_t k = bin.size() - 1;
209
210 bin.particleID(k) = a_leaf.particleID(i); // preserve id
211 bin.rankID(k) = dst.rank; // new owner
212 }
213 else {
214 outcast++;
215 }
216
217 a_leaf.remove(i); // swap-pop; do NOT advance i (a new particle now occupies slot i)
218 }
219
220 return outcast;
221 };
222
223#ifdef _OPENMP
224#pragma omp parallel reduction(+ : numOutcast)
225 {
226 MoverPool threadPool(numProc());
227
228#pragma omp for schedule(runtime)
229 for (int mybox = 0; mybox < nbox; mybox++) {
230 const DataIndex& din = dit[mybox];
231 const BaseFab<bool>* mask = (maskLevel != nullptr) ? &(*maskLevel)[din] : nullptr;
232
233 if (mask != nullptr && !mask->isUsable()) {
234 mask = nullptr;
235 }
236
237 numOutcast += partitionLeaf(threadPool, din, level[din], mask);
238 }
239
240#pragma omp critical(ParticleContainer_gatherMerge)
241 {
242 for (int r = 0; r < numProc(); r++) {
243 for (auto& kv : threadPool[r]) {
244 pool[r][kv.first].catenate(kv.second);
245 }
246 }
247 }
248 }
249#else
250 for (int mybox = 0; mybox < nbox; mybox++) {
251 const DataIndex& din = dit[mybox];
252 const BaseFab<bool>* mask = (maskLevel != nullptr) ? &(*maskLevel)[din] : nullptr;
253
254 if (mask != nullptr && !mask->isUsable()) {
255 mask = nullptr;
256 }
257
258 numOutcast += partitionLeaf(pool, din, level[din], mask);
259 }
260#endif
261 }
262
263 m_numOutcastLocal = numOutcast;
264
265 return pool;
266}

268template <typename P, typename Traits>
269inline void
270ParticleContainer<P, Traits>::distributeFromPool(MoverPool& a_pool, const bool a_setOwnerToReceiver)
271{
272 CH_TIME("ParticleContainer::distributeFromPool");
273
274 const int myRank = procID();
275 const int numRanks = numProc();
276
277 // ---- same-rank movers: catenate each pooled leaf into its destination leaf ----
278 // On the gather-all path the destination leaves are empty (remap clears them; regrid allocates them
279 // fresh) so catenate() takes the O(1) arena-swap fast path. On the mask fast path the destinations
280 // still hold their stayers, so catenate() appends (O(movers)); a mover never returns to its own
281 // origin leaf, so there is no self-append.
282 for (auto& kv : a_pool[myRank]) {
283 const int lvl = kv.first.first;
284 const unsigned int gidx = kv.first.second;
285 const DataIndex din = m_levelTiles[lvl]->getMyGrids().at(gidx);
286
287 (*m_particles[lvl])[din].catenate(kv.second);
288 }
289
290#ifdef CH_MPI
291 // ---- cross-rank movers: one Alltoallv of linearized particles ----
292 constexpr std::size_t pp = Leaf::bytesPerParticle();
293 constexpr std::size_t headerLen = 2 * sizeof(std::uint32_t) + sizeof(std::uint64_t);
294
295 // Size the send side before packing anything. A destination's byte count is arithmetic over the
296 // pool -- a header plus the payload for each pooled leaf -- so computing it up front costs nothing,
297 // and knowing it lets every particle be linearized straight into the flat buffer that MPI reads.
298 // Packing into per-destination buffers first and copying those into the flat one afterwards would
299 // hold a second full copy of every outbound particle for the duration of the copy.
300 std::vector<int> sendCounts(numRanks, 0);
301 std::vector<int> recvCounts(numRanks, 0);
302 std::vector<std::int64_t> sendBytes(numRanks, 0);
303
304 std::int64_t stot = 0;
305
306 for (int r = 0; r < numRanks; r++) {
307 if (r == myRank) {
308 continue;
309 }
310
311 std::int64_t bytes = 0;
312
313 for (const auto& kv : a_pool[r]) {
314 bytes += static_cast<std::int64_t>(headerLen + kv.second.size() * pp);
315 }
316
317 sendBytes[r] = bytes;
318 stot += bytes;
319 }
320
321 // MPI counts and displacements are int, so the byte totals have to fit in one. This is a real limit
322 // at scale rather than a formality -- a rank holding 2 GB of outbound particles reaches it -- and it
323 // has to be enforced in every build. CH_assert compiles out under OPT, which is precisely where the
324 // scale is, so a guard that used it would let the displacements overflow and corrupt the exchange
325 // silently instead of stopping it.
326 if (stot > static_cast<std::int64_t>(std::numeric_limits<int>::max())) {
327 MayDay::Abort("ParticleContainer::distributeFromPool - outbound particle bytes exceed MPI's int limit");
328 }
329
330 for (int r = 0; r < numRanks; r++) {
331 sendCounts[r] = static_cast<int>(sendBytes[r]);
332 }
333
334 MPI_Alltoall(sendCounts.data(), 1, MPI_INT, recvCounts.data(), 1, MPI_INT, Chombo_MPI::comm);
335
336 std::vector<int> sdispl(numRanks, 0);
337 std::vector<int> rdispl(numRanks, 0);
338
339 std::int64_t sacc = 0;
340 std::int64_t rtot = 0;
341
342 for (int r = 0; r < numRanks; r++) {
343 sdispl[r] = static_cast<int>(sacc);
344 sacc += sendCounts[r];
345
346 rdispl[r] = static_cast<int>(rtot);
347 rtot += recvCounts[r];
348 }
349
350 if (rtot > static_cast<std::int64_t>(std::numeric_limits<int>::max())) {
351 MayDay::Abort("ParticleContainer::distributeFromPool - inbound particle bytes exceed MPI's int limit");
352 }
353
354 std::vector<char> sflat(stot);
355 std::vector<char> rflat(rtot);
356
357 // The flat buffers are particle payload in flight, so account for them alongside the arenas. They
358 // are part of the peak this rank has to fit, and on a particle-heavy step they are comparable in
359 // size to the arenas themselves -- leaving them out would report a peak that understates the point
360 // in the step where an out-of-memory failure actually happens.
361 const std::size_t flatBytes = static_cast<std::size_t>(stot) + static_cast<std::size_t>(rtot);
362
363 ParticleMemory::addBytes(ParticleMemory::Kind::Buffer, flatBytes);
364
365 // Each destination's slice of sflat is disjoint and linearizeParticle is a const read, so packing
366 // different destinations in parallel is race-free. A pooled leaf is spent once it has been
367 // linearized, so release it there and then: holding it until the pool goes out of scope would keep
368 // a second copy of every outbound particle alive across the exchange itself, which is the point in
369 // the step where the footprint peaks.
370#ifdef _OPENMP
371#pragma omp parallel for schedule(runtime)
372#endif
373 for (int r = 0; r < numRanks; r++) {
374 if (r == myRank) {
375 continue;
376 }
377
378 char* p = sflat.data() + sdispl[r];
379
380 for (auto& kv : a_pool[r]) {
381 Leaf& src = kv.second;
382
383 const std::uint32_t lvl = static_cast<std::uint32_t>(kv.first.first);
384 const std::uint32_t gidx = static_cast<std::uint32_t>(kv.first.second);
385 const std::uint64_t count = static_cast<std::uint64_t>(src.size());
386
387 std::memcpy(p, &lvl, sizeof(std::uint32_t));
388 p += sizeof(std::uint32_t);
389
390 std::memcpy(p, &gidx, sizeof(std::uint32_t));
391 p += sizeof(std::uint32_t);
392
393 std::memcpy(p, &count, sizeof(std::uint64_t));
394 p += sizeof(std::uint64_t);
395
396 for (std::size_t i = 0; i < count; i++) {
397 src.linearizeParticle(p, i);
398 p += pp;
399 }
400
401 src.clear();
402 src.shrinkToFit();
403 }
404 }
405
406 MPI_Alltoallv(sflat.data(),
407 sendCounts.data(),
408 sdispl.data(),
409 MPI_BYTE,
410 rflat.data(),
411 recvCounts.data(),
412 rdispl.data(),
413 MPI_BYTE,
414 Chombo_MPI::comm);
415
416 // ---- unpack: group received chunks by destination leaf, then append in parallel ----
417 // Pass 1 (serial, header-only walk): record each chunk's payload pointer + count, grouped by its
418 // destination (level, gridIndex). Pass 2 appends in parallel with one destination leaf per thread,
419 // so no two threads ever touch the same leaf (delinearizeAndAppend mutates the leaf arena).
420 struct RecvChunk
421 {
422 const char* payload; // first particle of the chunk within rflat
423 std::uint64_t count;
424 };
425 struct DestGroup
426 {
427 int level;
428 unsigned int gridIndex;
429 std::uint64_t total;
430 std::vector<RecvChunk> chunks;
431 };
432
433 std::vector<DestGroup> groups;
434 std::map<PoolKey, std::size_t> keyToGroup;
435 {
436 const char* p = rflat.data();
437 const char* end = p + rtot;
438
439 while (p < end) {
440 std::uint32_t lvl;
441 std::uint32_t gidx;
442 std::uint64_t count;
443
444 std::memcpy(&lvl, p, sizeof(std::uint32_t));
445 p += sizeof(std::uint32_t);
446
447 std::memcpy(&gidx, p, sizeof(std::uint32_t));
448 p += sizeof(std::uint32_t);
449
450 std::memcpy(&count, p, sizeof(std::uint64_t));
451 p += sizeof(std::uint64_t);
452
453 const PoolKey key(static_cast<int>(lvl), gidx);
454 const auto it = keyToGroup.find(key);
455
456 if (it == keyToGroup.end()) {
457 keyToGroup.emplace(key, groups.size());
458 groups.push_back(DestGroup{static_cast<int>(lvl), gidx, count, {RecvChunk{p, count}}});
459 }
460 else {
461 DestGroup& g = groups[it->second];
462 g.total += count;
463 g.chunks.push_back(RecvChunk{p, count});
464 }
465
466 p += count * pp;
467 }
468 }
469
470 const int numGroups = static_cast<int>(groups.size());
471
472#ifdef _OPENMP
473#pragma omp parallel for schedule(runtime)
474#endif
475 for (int d = 0; d < numGroups; d++) {
476 const DestGroup& g = groups[d];
477 const DataIndex din = m_levelTiles[g.level]->getMyGrids().at(g.gridIndex);
478
479 Leaf& tgt = (*m_particles[g.level])[din];
480
481 tgt.reserve(tgt.size() + g.total); // one allocation; avoids geometric regrow during the appends
482 for (const RecvChunk& chunk : g.chunks) {
483 const char* q = chunk.payload;
484
485 for (std::uint64_t c = 0; c < chunk.count; c++) {
486 tgt.delinearizeAndAppend(q);
487 q += pp;
488 if (a_setOwnerToReceiver) {
489 tgt.rankID(tgt.size() - 1) = myRank; // remap/regrid: owner is the receiving rank
490 }
491 // else: keep the rankID carried in the buffer (ghost particles record their true owner)
492 }
493 }
494 }
495
496 ParticleMemory::removeBytes(ParticleMemory::Kind::Buffer, flatBytes);
497#endif
498}

500template <typename P, typename Traits>
501inline void
502ParticleContainer<P, Traits>::remap()
503{
504 CH_TIME("ParticleContainer::remap");
505
506 CH_assert(m_isDefined);
507
508 // Ghost particles are transient, non-owned copies; drop them so they are never routed as if owned.
509 this->clearGhostParticles();
510
511 if (m_validCells != nullptr && static_cast<int>(m_validCells->size()) >= m_finestLevel + 1) {
512 // Mask fast path: a particle still in its box and not under a finer grid cannot change owner, so we
513 // leave those "stayers" in place and pool only the movers. Destinations therefore keep their stayers
514 // -- do NOT clearParticles() here; distributeFromPool() appends the movers onto the existing leaves.
515 MoverPool pool = this->gatherMoversToPool();
516
517 this->distributeFromPool(pool, true); // remap: the receiving rank becomes the new owner
518 }
519 else {
520 // Fallback: re-pool every valid particle (used when no mask is available, e.g. addParticlesDestructive
521 // staging, or before the realm's valid-cell mask has been built).
522 MoverPool pool = this->gatherToPool(m_particles, m_grids);
523
524 this->clearParticles();
525 this->distributeFromPool(pool, true); // remap: the receiving rank becomes the new owner
526 }
527
528 m_isOrganizedByCell = false;
529}

546template <typename P, typename Traits>
547inline void
548ParticleContainer<P, Traits>::regrid(const Vector<DisjointBoxLayout>& a_grids,
549 const Vector<ProblemDomain>& a_domains,
550 const Vector<Real>& a_dx,
551 const Vector<int>& a_refRat,
552 const int a_minBlockSize,
553 const Vector<RefCountedPtr<LevelTiles>>& a_levelTiles,
554 const int a_newFinestLevel)
555{
556 CH_TIME("ParticleContainer::regrid");
557
558 CH_assert(m_isDefined);
559 CH_assert(a_levelTiles.size() > a_newFinestLevel);
560
561 // ---- adopt the new layout ----
562 m_grids = a_grids;
563 m_domains = a_domains;
564 m_refRat = a_refRat;
565 m_minBlockSize = a_minBlockSize;
566 m_finestLevel = a_newFinestLevel;
567
568 // Alias the Realm's per-level tile->box maps (the single source of truth); never build our own.
569 m_levelTiles = a_levelTiles;
570
571 m_dx.resize(a_newFinestLevel + 1);
572 for (int lvl = 0; lvl <= a_newFinestLevel; lvl++) {
573 m_dx[lvl] = a_dx[lvl] * RealVect::Unit;
574 }
575
576 // ---- re-allocate the (empty) holders over the new grids ----
577 m_particles.resize(a_newFinestLevel + 1);
578 m_maskParticles.resize(a_newFinestLevel + 1);
579 m_maskOrigins.resize(a_newFinestLevel + 1);
580
581 for (int lvl = 0; lvl <= a_newFinestLevel; lvl++) {
582 m_particles[lvl] = RefCountedPtr<LevelParticles>(new LevelParticles(a_grids[lvl]));
583 m_maskParticles[lvl] = RefCountedPtr<LevelParticles>(new LevelParticles(a_grids[lvl]));
584 m_maskOrigins[lvl] = RefCountedPtr<LayoutData<std::vector<std::size_t>>>(
585 new LayoutData<std::vector<std::size_t>>(a_grids[lvl]));
586 }
587
588 this->setupGrownGrids();
589
590 m_bufferParticles.resize(a_newFinestLevel + 1);
591
592 for (int lvl = 0; lvl <= a_newFinestLevel; lvl++) {
593 m_bufferParticles[lvl] = RefCountedPtr<LevelParticles>(new LevelParticles(m_grownGrids[lvl]));
594 }
595
596 // ---- redistribute the cached particles onto the new layout ----
597 MoverPool pool = this->gatherToPool(m_cacheParticles, m_cacheGrids);
598 this->distributeFromPool(pool, true); // regrid: the receiving rank becomes the new owner
599
600 // ---- release the cache ----
601 m_cacheParticles.resize(0);
602 m_cacheGrids.resize(0);
603
604 m_isOrganizedByCell = false;
605}

607template <typename P, typename Traits>
608inline void
609ParticleContainer<P, Traits>::clearGhostParticles(const GhostType a_type)
610{
611 CH_TIME("ParticleContainer::clearGhostParticles(GhostType)");
612
613 CH_assert(m_isDefined);
614 CH_assert(a_type != GhostType::Valid);
615
616 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
617 LevelParticles& level = *m_particles[lvl];
618 const DataIterator& dit = m_grids[lvl].dataIterator();
619
620 const int nbox = dit.size();
621
622#pragma omp parallel for schedule(runtime)
623 for (int mybox = 0; mybox < nbox; mybox++) {
624 Leaf& leaf = level[dit[mybox]];
625
626 // Swap-and-pop; the swapped-in particle now occupies index i, so do NOT advance on a hit.
627 std::size_t i = 0;
628 while (i < leaf.size()) {
629 if (leaf.ghost(i) == a_type) {
630 leaf.remove(i);
631 }
632 else {
633 i++;
634 }
635 }
636 }
637 }
638
639 m_isOrganizedByCell = false;
640}

642template <typename P, typename Traits>
643inline void
644ParticleContainer<P, Traits>::clearGhostParticles()
645{
646 CH_TIME("ParticleContainer::clearGhostParticles");
647
648 CH_assert(m_isDefined);
649
650 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
651 LevelParticles& level = *m_particles[lvl];
652 const DisjointBoxLayout& dbl = m_grids[lvl];
653 const DataIterator& dit = dbl.dataIterator();
654
655 const int nbox = dit.size();
656
657#pragma omp parallel for schedule(runtime)
658 for (int mybox = 0; mybox < nbox; mybox++) {
659 const DataIndex& din = dit[mybox];
660 Leaf& leaf = level[din];
661
662 // Swap-and-pop every ghost; the swapped-in particle now occupies index i, so do NOT advance on a hit.
663 std::size_t i = 0;
664 while (i < leaf.size()) {
665 if (leaf.isGhost(i)) {
666 leaf.remove(i);
667 }
668 else {
669 i++;
670 }
671 }
672 }
673 }
674
675 m_isOrganizedByCell = false; // swap-pop reorders the surviving valid particles

678template <typename P, typename Traits>
679inline void
680ParticleContainer<P, Traits>::claimOwnership(const bool a_onlyValidRegion)
681{
682 CH_TIME("ParticleContainer::claimOwnership");
683
684 CH_assert(m_isDefined);
685
686 const RankID myRank = static_cast<RankID>(procID());
687
688 // Ghosts are left untouched (they carry their true owner rank). Each non-ghost particle in the valid
689 // region (or all non-ghost particles when !a_onlyValidRegion) is stamped with this rank. Per-particle
690 // and box-local -> embarrassingly parallel.
691 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
692 LevelParticles& level = *m_particles[lvl];
693 const DisjointBoxLayout& dbl = m_grids[lvl];
694 const DataIterator& dit = dbl.dataIterator();
695 const int nbox = dit.size();
696
697#pragma omp parallel for schedule(runtime)
698 for (int mybox = 0; mybox < nbox; mybox++) {
699 const DataIndex& din = dit[mybox];
700 const Box box = dbl[din];
701 Leaf& leaf = level[din];
702
703 for (std::size_t i = 0; i < leaf.size(); i++) {
704 if (leaf.isGhost(i)) {
705 continue;
706 }
707 if (!a_onlyValidRegion || this->inValidBox(lvl, box, leaf.position(i))) {
708 leaf.rankID(i) = myRank;
709 }
710 }
711 }
712 }

715template <typename P, typename Traits>
716inline long long
717ParticleContainer<P, Traits>::resetParticleIDs(const bool a_onlyValidRegion, const ParticleID a_startID)
718{
719 CH_TIME("ParticleContainer::resetParticleIDs");
720
721 CH_assert(m_isDefined);
722
723 // A non-ghost particle qualifies for a new id iff (!a_onlyValidRegion) or its cell lies in the leaf's
724 // grid box. Ghosts never qualify (and are left untouched); non-qualifying non-ghost strays get
725 // s_invalidID. The predicate must be identical in the count and assign passes below.
726
727 // Flatten the local boxes in (level, box) order -- this fixes the local id ordering.
728 std::vector<std::pair<int, DataIndex>> boxes;
729 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
730 const DataIterator& dit = m_grids[lvl].dataIterator();
731 const int nbox = dit.size();
732 for (int mybox = 0; mybox < nbox; mybox++) {
733 boxes.emplace_back(lvl, dit[mybox]);
734 }
735 }
736 const int numBoxes = static_cast<int>(boxes.size());
737
738 // Pass 1: count qualifying particles per box (parallel; each box independent).
739 std::vector<long long> boxCount(numBoxes, 0);
740#pragma omp parallel for schedule(runtime)
741 for (int b = 0; b < numBoxes; b++) {
742 const int lvl = boxes[b].first;
743 const DataIndex& din = boxes[b].second;
744 const Box box = m_grids[lvl][din];
745 const Leaf& leaf = (*m_particles[lvl])[din];
746
747 long long count = 0;
748 for (std::size_t i = 0; i < leaf.size(); i++) {
749 if (!leaf.isGhost(i) && (!a_onlyValidRegion || this->inValidBox(lvl, box, leaf.position(i)))) {
750 count++;
751 }
752 }
753 boxCount[b] = count;
754 }
755
756 // Local exclusive prefix over boxes + local total.
757 std::vector<long long> boxLocalOffset(numBoxes, 0);
758 long long localTotal = 0;
759 for (int b = 0; b < numBoxes; b++) {
760 boxLocalOffset[b] = localTotal;
761 localTotal += boxCount[b];
762 }
763
764 // Per-rank exclusive offset: all-gather each rank's local total, then sum the ranks before this one.
765 const Vector<long long> perRankCount = ParallelOps::gather(localTotal);
766
767 ParticleID rankOffset = a_startID;
768 for (int r = 0; r < procID(); r++) {
769 rankOffset += perRankCount[r];
770 }
771
772 // Pass 2: assign ids (parallel; each box starts at its own offset). Ghosts untouched; qualifying get a
773 // contiguous id; non-qualifying non-ghost strays get s_invalidID.
774#pragma omp parallel for schedule(runtime)
775 for (int b = 0; b < numBoxes; b++) {
776 const int lvl = boxes[b].first;
777 const DataIndex& din = boxes[b].second;
778 const Box box = m_grids[lvl][din];
779 Leaf& leaf = (*m_particles[lvl])[din];
780
781 ParticleID id = rankOffset + static_cast<ParticleID>(boxLocalOffset[b]);
782 for (std::size_t i = 0; i < leaf.size(); i++) {
783 if (leaf.isGhost(i)) {
784 continue;
785 }
786 if (!a_onlyValidRegion || this->inValidBox(lvl, box, leaf.position(i))) {
787 leaf.particleID(i) = id++;
788 }
789 else {
790 leaf.particleID(i) = ParticleSoA<P, Traits>::s_invalidID;
791 }
792 }
793 }
794
795 return ParallelOps::sum(localTotal);
796}

798template <typename P, typename Traits>
799inline typename ParticleContainer<P, Traits>::MoverPool
800ParticleContainer<P, Traits>::gatherGhostsFromMasks(const AMRParticleGhostMask& a_maskSame,
801 const AMRParticleGhostMask& a_maskCoarToFine,
802 const AMRParticleGhostMask& a_maskFineToCoar)
803{
804 CH_TIME("ParticleContainer::gatherGhostsFromMasks");
805
806 MoverPool pool(numProc());
807
808 // Bin one source leaf's valid particles as ghosts of every target box listed by its cell's masks. The
809 // three masks are all indexed by the SOURCE cell on the source level, so a single cell lookup drives all
810 // directions. GhostType is from the RECEIVER's view: same level -> SameLevel, coarse source scattered to
811 // a finer patch -> Coarse, fine source scattered to a coarser patch -> Fine.
812 auto binLeaf = [&](MoverPool& a_target, const Leaf& a_leaf, const int a_srcLvl, const DataIndex& a_din) {
813 const bool haveSame = (a_srcLvl < a_maskSame.size()) && !a_maskSame[a_srcLvl].isNull();
814 const bool haveC2F = (a_srcLvl < m_finestLevel) && (a_srcLvl < a_maskCoarToFine.size()) &&
815 !a_maskCoarToFine[a_srcLvl].isNull();
816 const bool haveF2C = (a_srcLvl >= 1) && (a_srcLvl < a_maskFineToCoar.size()) &&
817 !a_maskFineToCoar[a_srcLvl].isNull();
818
819 const ParticleGhostMask* same = haveSame ? &((*a_maskSame[a_srcLvl])[a_din]) : nullptr;
820 const ParticleGhostMask* c2f = haveC2F ? &((*a_maskCoarToFine[a_srcLvl])[a_din]) : nullptr;
821 const ParticleGhostMask* f2c = haveF2C ? &((*a_maskFineToCoar[a_srcLvl])[a_din]) : nullptr;
822
823 if (same == nullptr && c2f == nullptr && f2c == nullptr) {
824 return;
825 }
826
827 for (std::size_t i = 0; i < a_leaf.size(); i++) {
828 const RealVect pos = a_leaf.position(i);
829 const IntVect iv = this->cellIndex(a_srcLvl, pos);
830
831 auto scatter =
832 [&](const ParticleGhostMask* a_mask, const int a_dstLvl, const GhostType a_gt, const bool a_pruneByTargetBox) {
833 if (a_mask == nullptr || !a_mask->box().contains(iv)) {
834 return;
835 }
836 const int nt = a_mask->numTargets(iv);
837
838 // Coarse-to-fine only: the source (coarse) cell is coarser than the fine ghost width, so a whole
839 // coarse cell would over-scatter. Keep the particle for a target only if its DESTINATION (fine)
840 // cell lies in that target's acceptance box (the exact fine ghost shell).
841 const IntVect dstCell = a_pruneByTargetBox ? this->cellIndex(a_dstLvl, pos) : IntVect::Zero;
842
843 for (int k = 0; k < nt; k++) {
844 if (a_pruneByTargetBox && !a_mask->targetBox(iv, k).contains(dstCell)) {
845 continue;
846 }
847
848 Leaf& bin = a_target[a_mask->targetRank(iv, k)][PoolKey(a_dstLvl, a_mask->targetGridIndex(iv, k))];
849
850 bin.append(pos, a_leaf.weight(i), a_leaf.gather(i));
851
852 const std::size_t j = bin.size() - 1;
853
854 bin.particleID(j) = a_leaf.particleID(i); // preserve id
855 bin.rankID(j) = procID(); // owner is this rank (it owns its valid particles)
856 bin.ghost(j) = a_gt;
857 }
858 };
859
860 scatter(same, a_srcLvl, GhostType::SameLevel, false);
861 scatter(c2f, a_srcLvl + 1, GhostType::Coarse, true);
862 scatter(f2c, a_srcLvl - 1, GhostType::Fine, false);
863 }
864 };
865
866 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
867 const LevelParticles& level = *m_particles[lvl];
868 const DataIterator& dit = m_grids[lvl].dataIterator();
869 const int nbox = dit.size();
870
871#ifdef _OPENMP
872#pragma omp parallel
873 {
874 MoverPool threadPool(numProc()); // thread-local; no locking while binning
875
876#pragma omp for schedule(runtime)
877 for (int mybox = 0; mybox < nbox; mybox++) {
878 binLeaf(threadPool, level[dit[mybox]], lvl, dit[mybox]);
879 }
880
881#pragma omp critical(ParticleContainer_gatherGhostsFromMasksMerge)
882 {
883 for (int r = 0; r < numProc(); r++) {
884 for (auto& kv : threadPool[r]) {
885 pool[r][kv.first].catenate(kv.second);
886 }
887 }
888 }
889 }
890#else
891 for (int mybox = 0; mybox < nbox; mybox++) {
892 binLeaf(pool, level[dit[mybox]], lvl, dit[mybox]);
893 }
894#endif
895 }
896
897 return pool;
898}

1235template <typename P, typename Traits>
1236template <typename Select>
1237inline bool
1238ParticleContainer<P, Traits>::moveToMaskHolder(Leaf& a_src,
1239 Leaf& a_dst,
1240 std::vector<std::size_t>& a_origins,
1241 const Select& a_select)
1242{
1243 a_origins.clear();
1244
1245 if (a_dst.size() != 0) {
1246 return false;
1247 }
1248
1249 const std::size_t numParticles = a_src.size();
1250
1251 for (std::size_t i = 0; i < numParticles; i++) {
1252 if (a_select(i)) {
1253 a_origins.push_back(i);
1254 }
1255 }
1256
1257 if (a_origins.empty()) {
1258 return true;
1259 }
1260
1261 const std::size_t numStay = numParticles - a_origins.size();
1262
1263 // Every particle moves, in its order.
1264 if (numStay == 0) {
1265 a_dst.catenate(a_src);
1266
1267 return true;
1268 }
1269
1270 // Put the selected particles at the end, keeping the order of both groups, and move them to the mask holder.
1271
1272 std::vector<std::size_t> dest(numParticles);
1273
1274 std::size_t stay = 0;
1275 std::size_t moved = numStay;
1276 std::size_t k = 0;
1277
1278 for (std::size_t i = 0; i < numParticles; i++) {
1279 if (k < a_origins.size() && a_origins[k] == i) {
1280 dest[i] = moved++;
1281 k++;
1282 }
1283 else {
1284 dest[i] = stay++;
1285 }
1286 }
1287
1288 a_src.permute(dest);
1289
1290 a_dst.reserve(a_origins.size());
1291
1292 for (std::size_t i = numStay; i < numParticles; i++) {
1293 a_dst.appendParticle(a_src, i);
1294 }
1295
1296 a_src.resize(numStay);
1297
1298 return true;
1299}

1303ParticleContainer<P, Traits>::returnMaskParticles()
1304{
1305 CH_TIME("ParticleContainer::returnMaskParticles");
1306
1307 CH_assert(m_isDefined);
1308
1309 for (int lvl = 0; lvl <= m_finestLevel; lvl++) {
1310 LevelParticles& maskLevel = *m_maskParticles[lvl];
1311 LevelParticles& srcLevel = *m_particles[lvl];
1312
1313 const DataIterator& dit = m_grids[lvl].dataIterator();
1314 const int nbox = dit.size();
1315
1316 // Set where a patch does not match its transfer. Aborting from inside the threaded loop is not safe.
1317 bool mismatch = false;
1318
1319#pragma omp parallel for schedule(runtime) reduction(|| : mismatch)
1320 for (int mybox = 0; mybox < nbox; mybox++) {
1321 const DataIndex& din = dit[mybox];
1322
1323 Leaf& src = srcLevel[din];
1324 Leaf& mask = maskLevel[din];
1325 std::vector<std::size_t>& origins = (*m_maskOrigins[lvl])[din];
1326
1327 // The mask holder must hold exactly the moved particles, and the recorded indices must fit the valid holder,
1328 // or the map below is not a permutation.
1329 if (mask.size() != origins.size()) {
1330 mismatch = true;
1331
1332 continue;
1333 }
1334
1335 if (origins.empty()) {
1336 continue;
1337 }
1338
1339 const std::size_t numStay = src.size();
1340 const std::size_t numParticles = numStay + origins.size();
1341
1342 if (origins.back() >= numParticles) {
1343 mismatch = true;
1344
1345 continue;
1346 }
1347
1348 // The particles that stayed are first, in their original order, and the returned ones follow in the order
1349 // they were moved out. Send each one back to its original index. If none stayed, that is already the order.
1350 src.catenate(mask);
1351
1352 if (numStay == 0) {
1353 origins.clear();
1354
1355 continue;
1356 }
1357
1358 std::vector<std::size_t> dest(numParticles);
1359
1360 std::size_t stay = 0;
1361 std::size_t k = 0;
1362
1363 for (std::size_t i = 0; i < numParticles; i++) {
1364 if (k < origins.size() && origins[k] == i) {
1365 dest[numStay + k] = i;
1366 k++;
1367 }
1368 else {
1369 dest[stay++] = i;
1370 }
1371 }
1372
1373 src.permute(dest);
1374
1375 origins.clear();
1376 }
1377
1378 if (mismatch) {
1379 MayDay::Abort("ParticleContainer::returnMaskParticles - the mask holder does not match the recorded indices");
1380 }
1381 }
1382
1383 // The order is restored, but the leaves were resized and appended to.
1384 m_isOrganizedByCell = false;
1385}
