42inline Real
43nnMergeDistSquared(const RealVect& a_a, const RealVect& a_b) noexcept
44{
45 Real d2 = 0.0;
46
47 for (int dir = 0; dir < SpaceDim; dir++) {
48 const Real delta = a_a[dir] - a_b[dir];
49
50 d2 += delta * delta;
51 }
52
53 return d2;
54}

63inline int
64nnMergeChebyshevCellDistance(const IntVect& a_a, const IntVect& a_b) noexcept
65{
66 int dist = 0;
67
68 for (int dir = 0; dir < SpaceDim; dir++) {
69 dist = std::max(dist, std::abs(a_a[dir] - a_b[dir]));
70 }
71
72 return dist;

91inline bool
92nnMergeCrossLevelTooFar(const RealVect& a_posA,
93 const int a_levelA,
94 const RealVect& a_posB,
95 const int a_levelB,
96 const RealVect& a_probLo,
97 const std::vector<RealVect>& a_dxByLevel,
98 const int a_maxCellDistance) noexcept
99{
100 const int refLevel = std::max(a_levelA, a_levelB);
101 const RealVect& refDx = a_dxByLevel[refLevel];
102 const IntVect keyA = cellKeyOf(a_posA, a_probLo, refDx);
103 const IntVect keyB = cellKeyOf(a_posB, a_probLo, refDx);
104 return nnMergeChebyshevCellDistance(keyA, keyB) > a_maxCellDistance;
105}

117inline void
118nnMergeWeightedCentroid(const RealVect& a_posA,
119 const Real a_weightA,
120 const RealVect& a_posB,
121 const Real a_weightB,
122 RealVect& a_mergedPos,
123 Real& a_mergedWeight) noexcept
124{
125 a_mergedWeight = a_weightA + a_weightB;
126
127 for (int dir = 0; dir < SpaceDim; dir++) {
128 a_mergedPos[dir] = (a_posA[dir] * a_weightA + a_posB[dir] * a_weightB) / a_mergedWeight;
129 }

144template <typename T>
145inline std::vector<T>
146nnMergeExchangeByRank(const std::vector<std::vector<T>>& a_sendByRank)
147{
148 const int numRanks = static_cast<int>(a_sendByRank.size());
149
150#ifdef CH_MPI
151 if (numRanks <= 1) {
152 return a_sendByRank.empty() ? std::vector<T>() : a_sendByRank[0];
153 }
154
155 std::vector<int> sendCounts(numRanks, 0);
156
157 for (int r = 0; r < numRanks; r++) {
158 sendCounts[r] = static_cast<int>(a_sendByRank[r].size());
159 }
160
161 std::vector<int> recvCounts(numRanks, 0);
162 MPI_Alltoall(sendCounts.data(), 1, MPI_INT, recvCounts.data(), 1, MPI_INT, Chombo_MPI::comm);
163
164 std::vector<int> sdispl(numRanks, 0);
165 std::vector<int> rdispl(numRanks, 0);
166
167 long stot = 0;
168 long rtot = 0;
169
170 for (int r = 0; r < numRanks; r++) {
171 sdispl[r] = static_cast<int>(stot);
172 stot += sendCounts[r];
173
174 rdispl[r] = static_cast<int>(rtot);
175 rtot += recvCounts[r];
176 }
177
178 std::vector<T> sflat(stot);
179
180 for (int r = 0; r < numRanks; r++) {
181 if (sendCounts[r] > 0) {
182 std::copy(a_sendByRank[r].begin(), a_sendByRank[r].end(), sflat.begin() + sdispl[r]);
183 }
184 }
185
186 std::vector<T> rflat(rtot);
187
188 // Scale record counts/displacements up to bytes -- MPI only knows bytes.
189 std::vector<int> sendBytes(numRanks);
190 std::vector<int> recvBytes(numRanks);
191 std::vector<int> sdisplBytes(numRanks);
192 std::vector<int> rdisplBytes(numRanks);
193
194 for (int r = 0; r < numRanks; r++) {
195 sendBytes[r] = sendCounts[r] * static_cast<int>(sizeof(T));
196 recvBytes[r] = recvCounts[r] * static_cast<int>(sizeof(T));
197 sdisplBytes[r] = sdispl[r] * static_cast<int>(sizeof(T));
198 rdisplBytes[r] = rdispl[r] * static_cast<int>(sizeof(T));
199 }
200
201 MPI_Alltoallv(sflat.data(),
202 sendBytes.data(),
203 sdisplBytes.data(),
204 MPI_BYTE,
205 rflat.data(),
206 recvBytes.data(),
207 rdisplBytes.data(),
208 MPI_BYTE,
209 Chombo_MPI::comm);
210
211 return rflat;
212#else
213 return a_sendByRank.empty() ? std::vector<T>() : a_sendByRank[0];
214#endif
215}

217inline IntVect
218cellKeyOf(const RealVect& a_position, const RealVect& a_probLo, const RealVect& a_dx) noexcept
219{
220 IntVect iv;
221
222 for (int dir = 0; dir < SpaceDim; dir++) {
223 iv[dir] = static_cast<int>(std::floor((a_position[dir] - a_probLo[dir]) / a_dx[dir]));
224 }
225
226 return iv;
227}

235template <typename Packed>
236inline void
237buildNNSpatialIndex(const std::vector<NNMergeParticle<Packed>>& a_localValid,
238 const std::vector<NNMergeParticle<Packed>>& a_ghosts,
239 const RealVect& a_probLo,
240 const RealVect& a_dx,
241 const int a_numParticlesPerCellThresh,
242 const int a_kNearest,
243 NNSpatialIndex<Packed>& a_index) noexcept
244{
245 CH_TIME("ParticleManagement::buildNNSpatialIndex");
246
247 a_index.pointCloud = nullptr;
248 a_index.pointCloudGraph.clear();
249 a_index.pointCloudNLocal = 0;
250 a_index.pointCloudK = a_kNearest;
251
252 const std::size_t total = a_localValid.size() + a_ghosts.size();
253 if (total == 0) {
254 return;
255 }
256
257 // Exact per-cell pre-filter (see this function's own docs): a count-only tally keyed as the
258 // crowding trigger (cellKeyOf(pos, probLo, dx), local and ghost alike). Build nothing unless
259 // some cell is over threshold. Checks the true per-cell maximum, not an average density (a
260 // low-average patch can still hold one over-threshold cell).
261 std::unordered_map<NNWalkCell, int, LevelTiles::TileHasher> cellCount;
262 cellCount.reserve(total);
263
264 for (const auto& p : a_localValid) {
265 cellCount[cellKeyOf(p.position, a_probLo, a_dx)]++;
266 }
267
268 for (const auto& p : a_ghosts) {
269 cellCount[cellKeyOf(p.position, a_probLo, a_dx)]++;
270 }
271
272 bool anyCellCrowded = false;
273
274 for (const auto& kv : cellCount) {
275 if (kv.second > a_numParticlesPerCellThresh) {
276 anyCellCrowded = true;
277 break;
278 }
279 }
280
281 if (!anyCellCrowded) {
282 return;
283 }
284
285 // Assemble the whole-patch point cloud (local valid first, then ghosts). Metadata is each
286 // point's globalID; a hit's cloud index decodes ownership (< nLocal => local valid, else
287 // ghost). Positions are embedded into 3D (z = 0 in 2D) for EBGeometry's Vec3T.
288 const std::size_t nLocal = a_localValid.size();
289
290 std::vector<EBGeometry::Vec3T<Real>> positions;
291 std::vector<ParticleID> metadata;
292
293 positions.reserve(total);
294 metadata.reserve(total);
295
296 auto pushPoint = [&](const NNMergeParticle<Packed>& a_p) {
297 EBGeometry::Vec3T<Real> v(0.0, 0.0, 0.0);
298
299 for (int dir = 0; dir < SpaceDim; dir++) {
300 v[dir] = a_p.position[dir];
301 }
302
303 positions.push_back(v);
304 metadata.push_back(a_p.globalID);
305 };
306
307 for (const auto& p : a_localValid) {
308 pushPoint(p);
309 }
310
311 for (const auto& p : a_ghosts) {
312 pushPoint(p);
313 }
314
315 a_index.pointCloud = std::make_shared<EBGeometry::PointCloudBVH<Real, ParticleID>>(positions, metadata);
316 a_index.pointCloudNLocal = nLocal;
317
318 // Precompute the whole kNN graph in one batched pass, reused across every internal
319 // a_iterateLocalTierToConvergence pass (positions never change within a round; only consumed
320 // status does, and that is filtered at lookup time). Ghost rows are computed but never read;
321 // see NNSpatialIndex::pointCloudGraph.
322 a_index.pointCloudGraph = a_index.pointCloud->allNearestNeighbors(static_cast<std::size_t>(a_kNearest));
323}

325template <typename Packed>
326inline void
327findNearestNeighborCandidates(const std::vector<NNMergeParticle<Packed>>& a_localValid,
328 const std::vector<NNMergeParticle<Packed>>& a_ghosts,
329 const std::unordered_set<ParticleID>& a_consumedIDs,
330 const NNSpatialIndex<Packed>& a_spatialIndex,
331 const std::unordered_map<NNCellKey, int, NNCellKeyHasher>& a_liveCellCount,
332 const RealVect& a_probLo,
333 const RealVect& a_dx,
334 const int a_numParticlesPerCellThresh,
335 std::vector<NNMergeEdge<Packed>>& a_edges,
336 const int a_maxFallbackCandidates,
337 const std::optional<int> a_maxCellDistance) noexcept
338{
339 CH_TIME("ParticleManagement::findNearestNeighborCandidates");
340
341 a_edges.clear();
342
343 const std::size_t nLocal = a_localValid.size();
344 const int K = 1 + a_maxFallbackCandidates;
345
346 // a_maxCellDistance's physical bound, computed once per call. The sqrt(SpaceDim) factor keeps
347 // this Euclidean prune no tighter than the per-axis Chebyshev cutoff applied downstream by
348 // resolveTrivialTier()/judgeProposals(), so it never excludes a candidate they would accept
349 // (see this function's own a_maxCellDistance docs).
350 std::optional<Real> maxDistBoundSquared;
351 if (a_maxCellDistance.has_value()) {
352 Real minDx = a_dx[0];
353
354 for (int dir = 1; dir < SpaceDim; dir++) {
355 minDx = std::min(minDx, a_dx[dir]);
356 }
357
358 const Real bound = static_cast<Real>(a_maxCellDistance.value()) * minDx * std::sqrt(static_cast<Real>(SpaceDim));
359 maxDistBoundSquared = bound * bound;
360 }
361
362 struct RankedCandidate
363 {
364 Real d2;
365 std::size_t idx;
366 bool isLocal;
367 };
368
369 for (std::size_t qi = 0; qi < nLocal; qi++) {
370 const NNMergeParticle<Packed>& q = a_localValid[qi];
371
372 // Lazy deletion -- a_localValid is the FULL per-patch set (see this function's own docs), so
373 // a query already consumed by an earlier pass/query this round is skipped explicitly here.
374 if (a_consumedIDs.count(q.globalID) > 0) {
375 continue;
376 }
377
378 // a_dx is this patch's own level's dx -- correct here since every query in a_localValid is
379 // necessarily on this same level (see this function's a_dx docs).
380 const IntVect qCell = cellKeyOf(q.position, a_probLo, a_dx);
381 const auto cntIt = a_liveCellCount.find(NNCellKey{q.level, qCell});
382 const int count = (cntIt != a_liveCellCount.end()) ? cntIt->second : 0;
383
384 if (count <= a_numParticlesPerCellThresh) {
385 continue;
386 }
387
388 std::vector<RankedCandidate> best;
389 best.reserve(static_cast<std::size_t>(K) + 1);
390
391 auto considerCandidate = [&](const Real a_d2, const std::size_t a_idx, const bool a_isLocal) {
392 if (static_cast<int>(best.size()) == K && a_d2 >= best.back().d2) {
393 return;
394 }
395
396 const auto insertPos = std::upper_bound(best.begin(),
397 best.end(),
398 a_d2,
399 [](const Real a_val, const RankedCandidate& a_c) {
400 return a_val < a_c.d2;
401 });
402
403 best.insert(insertPos, RankedCandidate{a_d2, a_idx, a_isLocal});
404
405 if (static_cast<int>(best.size()) > K) {
406 best.pop_back();
407 }
408 };
409
410 // Read this query's precomputed kNN row out of the PointCloudBVH graph (see
411 // buildNNSpatialIndex()). The query's own cloud index equals qi (local valid points packed
412 // first); allNearestNeighbors() already excluded it as self. Each hit's cloud index decodes
413 // ownership (< pointCloudNLocal => local valid, else ghost); apply the a_maxCellDistance bound
414 // and the a_consumedIDs skip before feeding survivors into the shared top-K logic. Filtering
415 // the distance-sorted k-nearest by the bound yields exactly the within-bound k-nearest.
416 // pointCloud is null only for a patch with no crowded cell, which generates no query.
417 if (a_spatialIndex.pointCloud) {
418 using PCBVH = EBGeometry::PointCloudBVH<Real, ParticleID>;
419
420 const std::size_t nLocalPC = a_spatialIndex.pointCloudNLocal;
421 const int Kgraph = a_spatialIndex.pointCloudK;
422 const std::size_t rowStart = qi * static_cast<std::size_t>(Kgraph);
423
424 for (int h = 0; h < Kgraph; h++) {
425 const typename PCBVH::Hit& hit = a_spatialIndex.pointCloudGraph[rowStart + static_cast<std::size_t>(h)];
426
427 // A hit with distanceSquared == max() is an empty slot (fewer than k neighbors existed).
428 if (hit.distanceSquared == std::numeric_limits<Real>::max()) {
429 break;
430 }
431 const std::size_t c = hit.index;
432 const bool isLocal = c < nLocalPC;
433 const std::size_t idx = isLocal ? c : c - nLocalPC;
434 const Real d2 = hit.distanceSquared;
435
436 if (maxDistBoundSquared.has_value() && d2 > maxDistBoundSquared.value()) {
437 continue;
438 }
439
440 const ParticleID gid = isLocal ? a_localValid[idx].globalID : a_ghosts[idx].globalID;
441
442 if (a_consumedIDs.count(gid) > 0) {
443 continue;
444 }
445
446 considerCandidate(d2, idx, isLocal);
447 }
448 }
449
450 if (best.empty()) {
451 continue;
452 }
453
454 NNMergeEdge<Packed> edge;
455
456 edge.distanceSquared = best[0].d2;
457 edge.queryID = q.globalID;
458 edge.candidateID = best[0].isLocal ? a_localValid[best[0].idx].globalID : a_ghosts[best[0].idx].globalID;
459 edge.candidateIsLocal = best[0].isLocal;
460
461 for (std::size_t k = 1; k < best.size(); k++) {
462 NNFallbackCandidate fc;
463
464 fc.candidateID = best[k].isLocal ? a_localValid[best[k].idx].globalID : a_ghosts[best[k].idx].globalID;
465 fc.distanceSquared = best[k].d2;
466 fc.candidateIsLocal = best[k].isLocal;
467
468 edge.fallbackCandidates.push_back(fc);
469 }
470
471 a_edges.push_back(std::move(edge));
472 }
473}

475template <typename Packed, typename Combine, typename PosValid, typename Allocator, typename OnMerge>
476inline void
477resolveTrivialTier(const std::vector<NNMergeEdge<Packed>>& a_edges,
478 const std::unordered_map<ParticleID, NNPooledParticle<Packed>>& a_particlesByID,
479 const Combine& a_combine,
480 std::unordered_map<NNCellKey, int, NNCellKeyHasher>& a_liveCellCount,
481 std::unordered_set<ParticleID>& a_consumedIDs,
482 const int a_numParticlesPerCellThresh,
483 std::unordered_set<ParticleID>& a_hasOutgoingCommitment,
484 std::vector<NNMergeResult<Packed>>& a_results,
485 const RealVect& a_probLo,
486 const std::vector<RealVect>& a_dxByLevel,
487 const PosValid& a_isPositionValid,
488 const Allocator& a_allocateID,
489 const int a_maxFallbackCandidates,
490 const std::optional<int> a_maxCellDistance,
491 unsigned long long* a_crossLevelMergeCount,
492 const OnMerge& a_onMergeCommitted) noexcept
493{
494 CH_TIME("ParticleManagement::resolveTrivialTier");
495
496 a_results.clear();
497
498 // 1. Sort a copy of a_edges by (distanceSquared, queryID, candidateID) -- see this function's
499 // docs for why the deterministic tiebreak is required, not optional.
500 std::vector<const NNMergeEdge<Packed>*> sorted;
501 sorted.reserve(a_edges.size());
502
503 for (const auto& e : a_edges) {
504 sorted.push_back(&e);
505 }
506
507 std::sort(sorted.begin(), sorted.end(), [](const NNMergeEdge<Packed>* a_a, const NNMergeEdge<Packed>* a_b) {
508 if (a_a->distanceSquared != a_b->distanceSquared) {
509 return a_a->distanceSquared < a_b->distanceSquared;
510 }
511
512 if (a_a->queryID != a_b->queryID) {
513 return a_a->queryID < a_b->queryID;
514 }
515
516 return a_a->candidateID < a_b->candidateID;
517 });
518
519 for (const NNMergeEdge<Packed>* edgePtr : sorted) {
520 const NNMergeEdge<Packed>& edge = *edgePtr;
521
522 // a. Stale query -- already consumed by an earlier, closer edge this same pass.
523 if (a_consumedIDs.count(edge.queryID) > 0) {
524 continue;
525 }
526
527 const auto queryDataIt = a_particlesByID.find(edge.queryID);
528
529 if (queryDataIt == a_particlesByID.end()) {
530 continue;
531 }
532
533 const NNMergeParticle<Packed>& query = queryDataIt->second.particle;
534 const bool queryExposed = queryDataIt->second.exposed;
535
536 bool resolved = false; // an outcome (commit, or a legitimate "leave unmatched") was reached
537 bool originalStale = false;
538 bool originalIneligible = false; // too far or invalid position, evaluated on the ORIGINAL candidate
539
540 const int numAttempts = a_maxFallbackCandidates + 1;
541
542 for (int attempt = 0; attempt < numAttempts; attempt++) {
543 // A boundary-exposed query can never merge trivially -- trivialOk requires !queryExposed
544 // unconditionally, so no fallback candidate could ever help. Run the ORIGINAL candidate
545 // (attempt 0) to set originalStale/originalIneligible, then stop rather than walking doomed
546 // fallbacks: each would cost consumed/busy lookups plus, for an alive candidate, a
547 // cross-level distance calc and an a_isPositionValid() call (an EB point query for EB-aware
548 // callers). Outcome is identical -- the query still proposes off its original candidate.
549 if (queryExposed && attempt > 0) {
550 break;
551 }
552
553 ParticleID candidateID;
554 bool candidateIsLocal;
555
556 if (attempt == 0) {
557 candidateID = edge.candidateID;
558 candidateIsLocal = edge.candidateIsLocal;
559 }
560 else {
561 const std::size_t fbIdx = static_cast<std::size_t>(attempt - 1);
562
563 if (fbIdx >= edge.fallbackCandidates.size()) {
564 break;
565 }
566
567 candidateID = edge.fallbackCandidates[fbIdx].candidateID;
568 candidateIsLocal = edge.fallbackCandidates[fbIdx].candidateIsLocal;
569 }
570
571 const bool candidateStale = candidateIsLocal && (a_consumedIDs.count(candidateID) > 0);
572 const bool candidateBusy = candidateIsLocal && !candidateStale &&
573 (a_hasOutgoingCommitment.count(candidateID) > 0);
574
575 const auto candDataIt = a_particlesByID.find(candidateID);
576 const bool candidateExists = candDataIt != a_particlesByID.end();
577
578 // exposed is meaningful only for local candidates (ghosts carry false); the single
579 // particlesByID lookup above now also yields it, so there is no separate exposed map.
580 const bool candidateExposed = candidateIsLocal && candidateExists && candDataIt->second.exposed;
581
582 bool tooFar = false;
583 bool invalidPos = false;
584
585 RealVect mergedPos(D_DECL(0.0, 0.0, 0.0));
586 Real mergedWeight = 0.0;
587
588 if (!candidateStale && candidateExists) {
589 const NNMergeParticle<Packed>& cand = candDataIt->second.particle;
590
591 if (a_maxCellDistance.has_value()) {
592 tooFar = nnMergeCrossLevelTooFar(query.position,
593 query.level,
594 cand.position,
595 cand.level,
596 a_probLo,
597 a_dxByLevel,
598 a_maxCellDistance.value());
599 }
600
601 if (!tooFar) {
602 nnMergeWeightedCentroid(query.position, query.weight, cand.position, cand.weight, mergedPos, mergedWeight);
603 invalidPos = !a_isPositionValid(mergedPos);
604 }
605 }
606
607 const bool ineligible = tooFar || invalidPos;
608
609 if (attempt == 0) {
610 originalStale = candidateStale;
611 originalIneligible = ineligible;
612 }
613
614 const bool trivialOk = candidateIsLocal && candidateExists && !candidateStale && !candidateBusy &&
615 !candidateExposed && !queryExposed && !ineligible;
616
617 if (trivialOk) {
618 const NNMergeParticle<Packed>& cand = candDataIt->second.particle;
619
620 // Each participant's OWN level's dx -- see a_dxByLevel's docs. The cell key must match
621 // the one a_liveCellCount was populated under for that participant, so a single shared
622 // dx would break the recheck for every level but one.
623 const NNCellKey qKey(query.level, cellKeyOf(query.position, a_probLo, a_dxByLevel[query.level]));
624 const NNCellKey cKey(cand.level, cellKeyOf(cand.position, a_probLo, a_dxByLevel[cand.level]));
625
626 const auto qCountIt = a_liveCellCount.find(qKey);
627 const auto cCountIt = a_liveCellCount.find(cKey);
628
629 const int qCount = (qCountIt != a_liveCellCount.end()) ? qCountIt->second : 0;
630 const int cCount = (cCountIt != a_liveCellCount.end()) ? cCountIt->second : 0;
631
632 if (qCount > a_numParticlesPerCellThresh && cCount > a_numParticlesPerCellThresh) {
633 // d. COMMIT.
634 NNMergeResult<Packed> result;
635
636 result.firstID = query.globalID;
637 result.secondID = cand.globalID;
638
639 result.merged.position = mergedPos;
640 result.merged.weight = mergedWeight;
641 result.merged.globalID = a_allocateID();
642 result.merged.ownerRank = query.ownerRank;
643
644 // Provisional -- placeMergedParticles() determines the TRUE destination level (and
645 // patch/rank) from the merged position via findDestination(), never from this field.
646 result.merged.level = query.level;
647 result.merged.payload = a_combine(query.payload, query.weight, cand.payload, cand.weight);
648
649 a_results.push_back(result);
650
651 a_consumedIDs.insert(query.globalID);
652 a_consumedIDs.insert(cand.globalID);
653
654 a_liveCellCount[qKey] -= 1;
655 a_liveCellCount[cKey] -= 1;
656
657 if (a_crossLevelMergeCount != nullptr && query.level != cand.level) {
658 (*a_crossLevelMergeCount)++;
659 }
660
661 a_onMergeCommitted(query, cand, result.merged);
662
663 resolved = true;
664 break;
665 }
666 else {
667 // e. Threshold recheck failed -- legitimately left unmatched, NOT fallback-eligible
668 // (a different candidate wouldn't change the query's own cell occupancy).
669 resolved = true;
670 break;
671 }
672 }
673 // else: fall through to the next fallback attempt, if any remain.
674 } // end attempt loop
675
676 if (!resolved) {
677 // f./g.: not resolved trivially -- see this function's docs for why only the ORIGINAL
678 // candidate's staleness/ineligibility (not any fallback's) decides whether a proposal is
679 // generated.
680 if (!originalStale && !originalIneligible) {
681 a_hasOutgoingCommitment.insert(query.globalID);
682 }
683 }
684 }

687template <typename Packed>
688inline void
689generateProposals(const std::vector<NNMergeEdge<Packed>>& a_edges,
690 const std::unordered_set<ParticleID>& a_hasOutgoingCommitment,
691 const std::unordered_map<ParticleID, NNPooledParticle<Packed>>& a_particlesByID,
692 const int a_numRanks,
693 std::vector<std::vector<NNMergeProposal<Packed>>>& a_proposalsByDestRank) noexcept
694{
695 CH_TIME("ParticleManagement::generateProposals");
696
697 a_proposalsByDestRank.clear();
698 a_proposalsByDestRank.resize(a_numRanks);
699
700 for (const auto& edge : a_edges) {
701 if (a_hasOutgoingCommitment.count(edge.queryID) == 0) {
702 continue;
703 }
704
705 const auto sourceIt = a_particlesByID.find(edge.queryID);
706 const auto targetIt = a_particlesByID.find(edge.candidateID);
707
708 if (sourceIt == a_particlesByID.end() || targetIt == a_particlesByID.end()) {
709 continue;
710 }
711
712 NNMergeProposal<Packed> proposal;
713
714 proposal.source = sourceIt->second.particle;
715 proposal.targetID = edge.candidateID;
716 proposal.distanceSquared = edge.distanceSquared;
717
718 const int destRank = static_cast<int>(targetIt->second.particle.ownerRank);
719
720 CH_assert(destRank >= 0 && destRank < a_numRanks);
721
722 a_proposalsByDestRank[destRank].push_back(proposal);
723 }
724}

726template <typename Packed, typename Combine, typename PosValid, typename Allocator, typename OnMerge>
727inline void
728judgeProposals(const std::vector<NNMergeProposal<Packed>>& a_incomingProposals,
729 const std::unordered_map<ParticleID, NNPooledParticle<Packed>>& a_particlesByID,
730 const std::unordered_set<ParticleID>& a_consumedIDs,
731 const std::unordered_set<ParticleID>& a_hasOutgoingCommitment,
732 const std::map<ParticleID, ParticleID>& a_outgoingTargetOf,
733 const Combine& a_combine,
734 std::unordered_map<NNCellKey, int, NNCellKeyHasher>& a_liveCellCount,
735 const int a_numParticlesPerCellThresh,
736 const RealVect& a_probLo,
737 const std::vector<RealVect>& a_dxByLevel,
738 const PosValid& a_isPositionValid,
739 const Allocator& a_allocateID,
740 std::vector<NNMergeResult<Packed>>& a_results,
741 std::vector<std::vector<NNMergeVerdict>>& a_verdictsByDestRank,
742 const std::optional<int> a_maxCellDistance,
743 unsigned long long* a_crossLevelMergeCount,
744 const OnMerge& a_onMergeCommitted) noexcept
745{
746 CH_TIME("ParticleManagement::judgeProposals");
747
748 a_results.clear();
749 // NOTE: a_verdictsByDestRank is APPENDED to, never resized here -- the caller must have
750 // already sized it to the total rank count (same convention as placeMergedParticles's
751 // a_scatterByDestRank).
752
753 // 1. Group incoming proposals by target id.
754 std::map<ParticleID, std::vector<const NNMergeProposal<Packed>*>> byTarget;
755
756 for (const auto& p : a_incomingProposals) {
757 byTarget[p.targetID].push_back(&p);
758 }
759
760 struct TargetDecision
761 {
762 ParticleID targetID;
763 const NNMergeProposal<Packed>* winner;
764 };
765
766 std::vector<TargetDecision> decisions;
767 decisions.reserve(byTarget.size());
768
769 for (auto& kv : byTarget) {
770 const ParticleID targetID = kv.first;
771 auto& proposals = kv.second;
772
773 const auto targetDataIt = a_particlesByID.find(targetID);
774
775 if (targetDataIt == a_particlesByID.end() || a_consumedIDs.count(targetID) > 0) {
776 // Target no longer alive -- defensive skip, reject every incoming proposal.
777 for (const NNMergeProposal<Packed>* p : proposals) {
778 const int destRank = static_cast<int>(p->source.ownerRank);
779 a_verdictsByDestRank[destRank].push_back(NNMergeVerdict{p->source.globalID, false});
780 }
781
782 continue;
783 }
784
785 const bool targetHasOutgoing = a_hasOutgoingCommitment.count(targetID) > 0;
786
787 const NNMergeProposal<Packed>* winner = nullptr;
788
789 if (!targetHasOutgoing) {
790 // 2. Argmin over incoming proposals, deterministic id tiebreak.
791 for (const NNMergeProposal<Packed>* p : proposals) {
792 if (winner == nullptr || p->distanceSquared < winner->distanceSquared ||
793 (p->distanceSquared == winner->distanceSquared && p->source.globalID < winner->source.globalID)) {
794 winner = p;
795 }
796 }
797 }
798 else {
799 // 3. Normally reject everything, UNLESS a mutual-match proposer is present.
800 const auto ownTargetIt = a_outgoingTargetOf.find(targetID);
801 if (ownTargetIt != a_outgoingTargetOf.end()) {
802 const ParticleID ownTarget = ownTargetIt->second;
803
804 for (const NNMergeProposal<Packed>* p : proposals) {
805 if (p->source.globalID == ownTarget) {
806 if (targetID < ownTarget) {
807 winner = p;
808 }
809
810 break;
811 }
812 }
813 }
814 }
815
816 if (winner != nullptr) {
817 decisions.push_back(TargetDecision{targetID, winner});
818 }
819
820 // Reject every OTHER proposal to this target immediately -- a rejection is never
821 // contingent on processing order, only an ACCEPT is (see rule 4).
822 for (const NNMergeProposal<Packed>* p : proposals) {
823 if (p != winner) {
824 const int destRank = static_cast<int>(p->source.ownerRank);
825
826 a_verdictsByDestRank[destRank].push_back(NNMergeVerdict{p->source.globalID, false});
827 }
828 }
829 }
830
831 // 4. Resolve all target decisions in ONE globally distance-sorted pass.
832 std::sort(decisions.begin(), decisions.end(), [](const TargetDecision& a_a, const TargetDecision& a_b) {
833 if (a_a.winner->distanceSquared != a_b.winner->distanceSquared) {
834 return a_a.winner->distanceSquared < a_b.winner->distanceSquared;
835 }
836
837 return a_a.targetID < a_b.targetID;
838 });
839
840 for (const TargetDecision& decision : decisions) {
841 const NNMergeProposal<Packed>& proposal = *decision.winner;
842 const auto targetIt = a_particlesByID.find(decision.targetID);
843
844 CH_assert(targetIt != a_particlesByID.end() && a_consumedIDs.count(decision.targetID) == 0);
845
846 const NNMergeParticle<Packed>& target = targetIt->second.particle;
847 const NNMergeParticle<Packed>& source = proposal.source;
848
849 const int sourceDestRank = static_cast<int>(source.ownerRank);
850
851 // 6. a_maxCellDistance / a_isPositionValid: a physical-validity rule, applies here exactly
852 // as in resolveTrivialTier(), regardless of the mutual-match exception above.
853 bool tooFar = false;
854
855 if (a_maxCellDistance.has_value()) {
856 tooFar = nnMergeCrossLevelTooFar(target.position,
857 target.level,
858 source.position,
859 source.level,
860 a_probLo,
861 a_dxByLevel,
862 a_maxCellDistance.value());
863 }
864
865 if (tooFar) {
866 a_verdictsByDestRank[sourceDestRank].push_back(NNMergeVerdict{source.globalID, false});
867 continue;
868 }
869
870 RealVect mergedPos(D_DECL(0.0, 0.0, 0.0));
871 Real mergedWeight = 0.0;
872
873 nnMergeWeightedCentroid(target.position, target.weight, source.position, source.weight, mergedPos, mergedWeight);
874
875 if (!a_isPositionValid(mergedPos)) {
876 a_verdictsByDestRank[sourceDestRank].push_back(NNMergeVerdict{source.globalID, false});
877
878 continue;
879 }
880
881 // Dynamic crowding recheck for BOTH participants (mirrors resolveTrivialTier()'s symmetric
882 // recheck -- each participant's OWN level's dx, see a_dxByLevel's docs). The source's cell
883 // must be rechecked too: an earlier merge this round can drain it below threshold between
884 // proposal generation and judging, and accepting anyway would violate the crowding trigger on
885 // the source side. If this rank has no live entry for the source's cell (never shipped here as
886 // a ghost), sCount defaults to 0 -- rejected -- the same conservative default used for an
887 // unknown target count.
888 const NNCellKey tKey(target.level, cellKeyOf(target.position, a_probLo, a_dxByLevel[target.level]));
889 const NNCellKey sKey(source.level, cellKeyOf(source.position, a_probLo, a_dxByLevel[source.level]));
890
891 const auto tCountIt = a_liveCellCount.find(tKey);
892 const auto sCountIt = a_liveCellCount.find(sKey);
893
894 const int tCount = (tCountIt != a_liveCellCount.end()) ? tCountIt->second : 0;
895 const int sCount = (sCountIt != a_liveCellCount.end()) ? sCountIt->second : 0;
896
897 if (tCount <= a_numParticlesPerCellThresh || sCount <= a_numParticlesPerCellThresh) {
898 a_verdictsByDestRank[sourceDestRank].push_back(NNMergeVerdict{source.globalID, false});
899
900 continue;
901 }
902
903 // Accept.
904 NNMergeResult<Packed> result;
905
906 result.firstID = target.globalID;
907 result.secondID = source.globalID;
908
909 result.merged.position = mergedPos;
910 result.merged.weight = mergedWeight;
911 result.merged.globalID = a_allocateID();
912 result.merged.ownerRank = target.ownerRank;
913
914 // Provisional -- placeMergedParticles() determines the TRUE destination level (and
915 // patch/rank) from the merged position via findDestination(), never from this field.
916 result.merged.level = target.level;
917 result.merged.payload = a_combine(target.payload, target.weight, source.payload, source.weight);
918
919 a_results.push_back(result);
920
921 // Symmetric decrement (mirrors resolveTrivialTier()'s commit), but only when this rank has a
922 // live entry for the source's cell (sCountIt found above); otherwise there is nothing local
923 // to decrement -- the source's owning rank updates its own copy on the accepted verdict.
924 a_liveCellCount[tKey] -= 1;
925
926 if (sCountIt != a_liveCellCount.end()) {
927 a_liveCellCount[sKey] -= 1;
928 }
929
930 if (a_crossLevelMergeCount != nullptr && target.level != source.level) {
931 (*a_crossLevelMergeCount)++;
932 }
933
934 a_onMergeCommitted(target, source, result.merged);
935
936 a_verdictsByDestRank[sourceDestRank].push_back(NNMergeVerdict{source.globalID, true});
937 }
938}

953template <typename Packed>
954inline void
955placeMergedParticles(const std::vector<NNMergeResult<Packed>>& a_results,
956 const PositionLocator& a_locate,
957 const int a_thisRank,
958 std::map<std::pair<int, unsigned int>, std::vector<NNMergeParticle<Packed>>>& a_localByPatch,
959 std::vector<std::vector<NNMergeParticle<Packed>>>& a_scatterByDestRank) noexcept
960{
961 CH_TIME("ParticleManagement::placeMergedParticles");
962
963 a_localByPatch.clear();
964 // a_scatterByDestRank is APPENDED to, never resized here -- see its own docs.
965
966 for (const auto& result : a_results) {
967 // O(1) point->block query -- see PositionLocator/NNParticleLocation. Never a linear scan over
968 // patches (the bottleneck this avoids at large patch counts).
969 const NNParticleLocation loc = a_locate(result.merged.position);
970
971 // A merged particle landing outside every known patch (e.g. exactly on/outside the domain
972 // boundary due to floating-point roundoff in the weighted centroid) is not expected in
973 // normal operation; MayDay rather than silently dropping mass.
974 if (!loc.valid) {
975 MayDay::Error("ParticleManagement::placeMergedParticles -- merged particle position not "
976 "found in any patch");
977 }
978
979 if (loc.ownerRank == a_thisRank) {
980 a_localByPatch[std::make_pair(loc.level, loc.gridIndex)].push_back(result.merged);
981 }
982 else {
983 a_scatterByDestRank[loc.ownerRank].push_back(result.merged);
984 }
985 }
986}

990template <typename P,
991 typename Packed,
992 typename Traits,
993 typename Gather,
994 typename Combine,
995 typename Scatter,
996 typename Allocator,
997 typename PosValid,
998 typename OnMerge>
999inline void
1000mergeNearestNeighborsRound(AmrMesh& a_amr,
1001 ParticleContainer<P, Traits>& a_particles,
1002 const int a_numParticlesPerCellThresh,
1003 const Gather& a_gather,
1004 const Combine& a_combine,
1005 const Scatter& a_scatter,
1006 const Allocator& a_allocateID,
1007 const bool a_iterateLocalTierToConvergence,
1008 const int a_maxFallbackCandidates,
1009 const std::optional<int> a_maxCellDistance,
1010 const int a_ghostWidth,
1011 const PosValid& a_isPositionValid,
1012 unsigned long long* a_crossLevelMergeCount,
1013 const OnMerge& a_onMergeCommitted)
1014{
1015 CH_TIME("ParticleManagement::mergeNearestNeighborsRound");
1016
1017 // This function is the public orchestrator; the step helpers and their types live in detail::.
1018 using namespace detail;
1019
1020 const int myRank = procID();
1021 const int numRanks = static_cast<int>(numProc());
1022
1023 const std::string realm = a_particles.getRealm();
1024 const int finestLevel = a_amr.getFinestLevel();
1025 const RealVect probLo = a_amr.getProbLo();
1026 const Vector<Real>& dxScalar = a_amr.getDx();
1027
1028 // Ghost particles are assumed ALREADY FILLED by the caller (an a_ghostWidth-cell halo -- one cell
1029 // per unit of a_ghostWidth in every direction, which is what the crowding trigger and the candidate
1030 // search look at) and are used here as candidates. This function does not fill them: filling is the
1031 // caller's responsibility, on equal footing with the caller cell-sorting before a cell-based merge
1032 // (see ItoSolver / ParticleManagement::ParticleMergeKind). A particle ghost mask of this same width
1033 // must have been registered on this realm before the grids were (re)built, and every rank must use
1034 // the same a_ghostWidth (see the a_ghostWidth docs -- it defines boundary-exposure).
1035
1036 struct PatchWork
1037 {
1038 int level;
1039 DataIndex din;
1040 RealVect dx;
1041 };
1042
1043 std::vector<PatchWork> patchWork;
1044 std::vector<std::vector<NNMergeParticle<Packed>>> patchLocalValid; // parallel to patchWork
1045 std::vector<std::vector<NNMergeParticle<Packed>>> patchGhosts; // parallel to patchWork
1046 std::vector<NNSpatialIndex<Packed>> patchSpatialIndex; // parallel to patchWork
1047
1048 // pooled over this rank's own patches; each value carries the particle plus its boundary-exposure
1049 // flag (folded in from what used to be a separate exposed map, halving the gather's per-particle
1050 // inserts and resolveTrivialTier()'s per-query/candidate lookups)
1051 std::unordered_map<ParticleID, NNPooledParticle<Packed>> particlesByID;
1052 std::unordered_map<NNCellKey, int, NNCellKeyHasher> liveCellCount; // pooled, level-qualified cell key
1053
1054 // Per-level dx, indexed by level -- every cell-key computation below and downstream (in
1055 // resolveTrivialTier()/judgeProposals()) MUST use the dx of the PARTICLE'S OWN level (see
1056 // NNMergeParticle::level's docs), never a single dx shared across levels.
1057 std::vector<RealVect> dxByLevel(finestLevel + 1);
1058
1059 for (int lvl = 0; lvl <= finestLevel; lvl++) {
1060 dxByLevel[lvl] = RealVect(D_DECL(dxScalar[lvl], dxScalar[lvl], dxScalar[lvl]));
1061 }
1062
1063 CH_assert(a_ghostWidth >= 1);
1064 const int ghostWidth = a_ghostWidth;
1065
1066 for (int lvl = 0; lvl <= finestLevel; lvl++) {
1067 const RealVect& dx = dxByLevel[lvl];
1068
1069 const DisjointBoxLayout& dbl = a_amr.getGrids(realm)[lvl];
1070
1071 const AMRParticleGhostMask& maskSame = a_amr.getParticleGhostMask(realm, ghostWidth);
1072 const AMRParticleGhostMask& maskC2F = a_amr.getParticleGhostMaskCoarToFine(realm, ghostWidth);
1073 const AMRParticleGhostMask& maskF2C = a_amr.getParticleGhostMaskFineToCoar(realm, ghostWidth);
1074
1075 const DataIterator& dit = dbl.dataIterator();
1076 const int nbox = dit.size();
1077
1078 for (int mybox = 0; mybox < nbox; mybox++) {
1079 const DataIndex& din = dit[mybox];
1080
1081 ParticleSoA<P, Traits>& leaf = a_particles[lvl][din];
1082
1083 std::vector<NNMergeParticle<Packed>> localValid;
1084 std::vector<NNMergeParticle<Packed>> ghosts;
1085
1086 localValid.reserve(leaf.size());
1087 ghosts.reserve(leaf.size());
1088
1089 // Reserve to the running total -- particlesByID is pooled across every patch this rank owns,
1090 // so reserving up front avoids repeated rehashes as it grows. leaf.size() is O(1).
1091 particlesByID.reserve(particlesByID.size() + leaf.size());
1092
1093 // Boundary-exposure masks for THIS patch (din), resolved ONCE here rather than per particle:
1094 // din/lvl are fixed within this box, so the RefCountedPtr null-checks and the LayoutData[din]
1095 // lookups are loop-invariant. A null pointer means the mask does not apply at this level (no
1096 // coarser/finer neighbour, or none defined).
1097 const ParticleGhostMask* maskSameDin = maskSame[lvl].isNull() ? nullptr : &(*maskSame[lvl])[din];
1098 const ParticleGhostMask* maskC2FDin = (lvl < finestLevel && !maskC2F[lvl].isNull()) ? &(*maskC2F[lvl])[din]
1099 : nullptr;
1100 const ParticleGhostMask* maskF2CDin = (lvl > 0 && !maskF2C[lvl].isNull()) ? &(*maskF2C[lvl])[din] : nullptr;
1101
1102 for (std::size_t i = 0; i < leaf.size(); i++) {
1103 const bool isGhost = leaf.isGhost(i);
1104
1105 // A ghost's TRUE origin level is NOT this receiving patch's level `lvl` -- it is
1106 // lvl-1/lvl+1 for a Coarse/Fine ghost (see ParticleSoA::GhostType). Its cell key must be
1107 // computed with that origin level's dx (see NNMergeParticle::level).
1108 int originLevel = lvl;
1109 if (isGhost) {
1110 switch (leaf.ghost(i)) {
1111 case GhostType::Coarse: {
1112 originLevel = lvl - 1;
1113
1114 break;
1115 }
1116 case GhostType::Fine: {
1117 originLevel = lvl + 1;
1118
1119 break;
1120 }
1121 default: {
1122 originLevel = lvl; // GhostType::SameLevel
1123
1124 break;
1125 }
1126 }
1127 }
1128
1129 NNMergeParticle<Packed> p;
1130
1131 p.position = leaf.position(i);
1132 p.weight = leaf.weight(i);
1133 p.globalID = leaf.particleID(i);
1134 p.ownerRank = leaf.rankID(i);
1135 p.level = originLevel;
1136 p.payload = a_gather(leaf, i);
1137
1138 const IntVect cell = cellKeyOf(p.position, probLo, dxByLevel[originLevel]);
1139
1140 if (isGhost) {
1141 ghosts.push_back(p);
1142 // REQUIRED -- generateProposals()'s a_particlesByID contract is "every currently-alive
1143 // local valid particle AND every ghost". A ghost whose true owning patch is not among
1144 // this rank's own patches has no native entry anywhere in this loop, so without this
1145 // insert generateProposals()'s lookup fails and the proposal is dropped. Ghosts are never
1146 // exposed (a ghost never initiates a query).
1147 particlesByID[p.globalID] = NNPooledParticle<Packed>{p, false};
1148 }
1149 else {
1150 bool isExposed = (maskSameDin != nullptr) && isBoundaryExposed(cell, *maskSameDin);
1151
1152 if (!isExposed && maskC2FDin != nullptr) {
1153 isExposed = isBoundaryExposed(cell, *maskC2FDin);
1154 }
1155
1156 if (!isExposed && maskF2CDin != nullptr) {
1157 isExposed = isBoundaryExposed(cell, *maskF2CDin);
1158 }
1159
1160 localValid.push_back(p);
1161 particlesByID[p.globalID] = NNPooledParticle<Packed>{p, isExposed};
1162 }
1163 }
1164
1165 // Built once here, per patch, per round -- over the FULL localValid/ghosts sets, before any
1166 // consumption this round -- and reused across every internal a_iterateLocalTierToConvergence
1167 // pass via lazy deletion (see findNearestNeighborCandidates()'s docs).
1168 NNSpatialIndex<Packed> spatialIndex;
1169
1170 buildNNSpatialIndex(localValid,
1171 ghosts,
1172 probLo,
1173 dx,
1174 a_numParticlesPerCellThresh,
1175 1 + a_maxFallbackCandidates,
1176 spatialIndex);
1177
1178 patchWork.push_back(PatchWork{lvl, din, dx});
1179 patchLocalValid.push_back(std::move(localValid));
1180 patchGhosts.push_back(std::move(ghosts));
1181 patchSpatialIndex.push_back(std::move(spatialIndex));
1182 }
1183 }
1184
1185 // Populate liveCellCount from particlesByID, NOT inline in the gather loop above. A single
1186 // particle can appear as a ghost in several of this rank's own patches at once, so counting per
1187 // occurrence would multiply-count it. particlesByID is keyed by globalID, so those repeated
1188 // occurrences collapse to one entry -- deriving the count from it counts each particle once.
1189 for (const auto& kv : particlesByID) {
1190 const NNMergeParticle<Packed>& p = kv.second.particle;
1191 const IntVect cell = cellKeyOf(p.position, probLo, dxByLevel[p.level]);
1192
1193 liveCellCount[NNCellKey(p.level, cell)] += 1;
1194 }
1195
1196 std::unordered_set<ParticleID> consumedIDs;
1197 std::unordered_set<ParticleID> hasOutgoingCommitment;
1198 std::map<ParticleID, ParticleID> outgoingTargetOf;
1199
1200 std::vector<NNMergeResult<Packed>> allResults;
1201
1202 // 2. Per-patch candidate search + trivial-tier resolution, pooled rank-wide.
1203 //
1204 // lastPooledEdges deliberately survives past the loop below: it is reused (NOT recomputed) for
1205 // outgoingTargetOf/generateProposals afterward. Nothing consumes any particles between the last
1206 // runLocalTierPass() and generateProposals(), so the edges are still exactly current; re-running
1207 // the candidate search purely to re-derive them would repeat the most expensive step for no new
1208 // information.
1209 std::vector<NNMergeEdge<Packed>> lastPooledEdges;
1210
1211 auto runLocalTierPass = [&]() -> bool {
1212 lastPooledEdges.clear();
1213
1214 for (std::size_t pw = 0; pw < patchWork.size(); pw++) {
1215 // patchLocalValid[pw]/patchGhosts[pw] are this patch's FULL sets and patchSpatialIndex[pw]
1216 // was built once over them before the local-tier passes began. findNearestNeighborCandidates()
1217 // filters against consumedIDs itself (see its docs), so nothing is rebuilt or re-filtered here.
1218 std::vector<NNMergeEdge<Packed>> patchEdges;
1219
1220 findNearestNeighborCandidates(patchLocalValid[pw],
1221 patchGhosts[pw],
1222 consumedIDs,
1223 patchSpatialIndex[pw],
1224 liveCellCount,
1225 probLo,
1226 patchWork[pw].dx,
1227 a_numParticlesPerCellThresh,
1228 patchEdges,
1229 a_maxFallbackCandidates,
1230 a_maxCellDistance);
1231
1232 lastPooledEdges.insert(lastPooledEdges.end(), patchEdges.begin(), patchEdges.end());
1233 }
1234
1235 std::vector<NNMergeResult<Packed>> passResults;
1236
1237 resolveTrivialTier(lastPooledEdges,
1238 particlesByID,
1239 a_combine,
1240 liveCellCount,
1241 consumedIDs,
1242 a_numParticlesPerCellThresh,
1243 hasOutgoingCommitment,
1244 passResults,
1245 probLo,
1246 dxByLevel,
1247 a_isPositionValid,
1248 a_allocateID,
1249 a_maxFallbackCandidates,
1250 a_maxCellDistance,
1251 a_crossLevelMergeCount,
1252 a_onMergeCommitted);
1253
1254 allResults.insert(allResults.end(), passResults.begin(), passResults.end());
1255
1256 return !passResults.empty();
1257 };
1258
1259 if (a_iterateLocalTierToConvergence) {
1260 constexpr int safetyCap = 1000;
1261
1262 for (int iter = 0; iter < safetyCap; iter++) {
1263 // hasOutgoingCommitment must NOT accumulate across internal iterations -- each iteration
1264 // re-derives it from scratch via a fresh candidate search over the currently-alive set.
1265 hasOutgoingCommitment.clear();
1266
1267 if (!runLocalTierPass()) {
1268 break;
1269 }
1270 }
1271 }
1272 else {
1273 runLocalTierPass();
1274 }
1275
1276 // outgoingTargetOf: for every query id with an outgoing commitment, remember its own nearest-
1277 // neighbor candidate id (needed by judgeProposals()'s mutual-match rule). Reuses lastPooledEdges
