104template <typename T>
105void
106EBAMRData<T>::exchange() noexcept
107{
108 CH_TIMERS("EBAMRData<T>::exchange()");
109 CH_TIMER("EBAMRData<T>::exchange()::post", t1);
110 CH_TIMER("EBAMRData<T>::exchange()::local", t2);
111 CH_TIMER("EBAMRData<T>::exchange()::wait", t3);
112
113 // Pipeline the hierarchy instead of running one complete MPI round per level. Every level's messages go onto
114 // the wire first, then the on-rank copies run while they are in flight, and only then does anything wait. The
115 // exposed cost is the slowest single level rather than the sum over levels, and no level's messages are held
116 // back by the on-rank work of the levels below it.
117 //
118 // This is safe because the send and receive buffers, and the MPI requests, live on the Copier -- and each
119 // level's LevelData owns its own exchange Copier. The levels therefore never alias one another's buffers.
120 // Message matching is unchanged as well: every rank still posts its levels in the same order, so MPI's
121 // non-overtaking guarantee pairs sends with receives exactly as it does when the levels run one at a time.
122 //
123 // The IntVect::Zero test is the one LevelData::exchange() makes for itself. A level with no ghost cells has
124 // nothing to exchange, so it must be skipped here rather than handed to exchangePost().
125 const int numLevels = m_data.size();
126
127 CH_START(t1);
128 for (int lvl = 0; lvl < numLevels; lvl++) {
129 if (m_data[lvl]->ghostVect() != IntVect::Zero) {
130 m_data[lvl]->exchangePost(m_data[lvl]->exchangeCopier());
131 }
132 }
133 CH_STOP(t1);
134
135 CH_START(t2);
136 for (int lvl = 0; lvl < numLevels; lvl++) {
137 if (m_data[lvl]->ghostVect() != IntVect::Zero) {
138 m_data[lvl]->exchangeLocal(m_data[lvl]->exchangeCopier());
139 }
140 }
141 CH_STOP(t2);
142
143 CH_START(t3);
144 for (int lvl = 0; lvl < numLevels; lvl++) {
145 if (m_data[lvl]->ghostVect() != IntVect::Zero) {
146 m_data[lvl]->exchangeEnd();
147 }
148 }
149 CH_STOP(t3);
150}
