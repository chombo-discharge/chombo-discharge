372class NNCellBudget
373{
374public:
379 NNCellBudget(const int a_uniform) noexcept : m_uniform(a_uniform)
380 {}
381
388 inline void
389 reserveOne(const int a_level, const IntVect& a_cell)
390 {
391 const int count = ++m_reserved[detail::NNCellKey(a_level, a_cell)];
392
393 m_minBudget = std::min(m_minBudget, std::max(0, m_uniform - count));
394 }
395
405 inline void
406 reserveOne(const int a_level, const RealVect& a_position, const RealVect& a_probLo, const RealVect& a_dx)
407 {
408 this->reserveOne(a_level, detail::cellKeyOf(a_position, a_probLo, a_dx));
409 }
410
416 inline int
417 operator()(const detail::NNCellKey& a_key) const noexcept
418 {
419 if (m_reserved.empty()) {
420 return m_uniform;
421 }
422
423 const auto it = m_reserved.find(a_key);
424
425 return (it == m_reserved.end()) ? m_uniform : std::max(0, m_uniform - it->second);
426 }
427
435 inline int
436 minimum() const noexcept
437 {
438 return m_reserved.empty() ? m_uniform : std::min(m_uniform, m_minBudget);
439 }
440
441private:
445 int m_uniform;
446
450 int m_minBudget = std::numeric_limits<int>::max();
451
455 std::unordered_map<detail::NNCellKey, int, detail::NNCellKeyHasher> m_reserved;
456};
