424class NNCellBudget
425{
426public:
431 NNCellBudget(const int a_uniform) noexcept : m_uniform(a_uniform)
432 {}
433
440 inline void
441 reserveOne(const int a_level, const IntVect& a_cell)
442 {
443 const int count = ++m_reserved[detail::NNCellKey(a_level, a_cell)];
444
445 m_minBudget = std::min(m_minBudget, std::max(0, m_uniform - count));
446 }
447
457 inline void
458 reserveOne(const int a_level, const RealVect& a_position, const RealVect& a_probLo, const RealVect& a_dx)
459 {
460 this->reserveOne(a_level, detail::cellKeyOf(a_position, a_probLo, a_dx));
461 }
462
468 inline int
469 operator()(const detail::NNCellKey& a_key) const noexcept
470 {
471 if (m_reserved.empty()) {
472 return m_uniform;
473 }
474
475 const auto it = m_reserved.find(a_key);
476
477 return (it == m_reserved.end()) ? m_uniform : std::max(0, m_uniform - it->second);
478 }
479
487 inline int
488 minimum() const noexcept
489 {
490 return m_reserved.empty() ? m_uniform : std::min(m_uniform, m_minBudget);
491 }
492
493private:
497 int m_uniform;
498
502 int m_minBudget = std::numeric_limits<int>::max();
503
507 std::unordered_map<detail::NNCellKey, int, detail::NNCellKeyHasher> m_reserved;
508};
