1623 std::size_t
1624 cellKey(const std::size_t a_index, const Box& a_box, const RealVect& a_dx, const RealVect& a_probLo) const noexcept
1625 {
1626 const IntVect loEnd = a_box.smallEnd();
1627 const IntVect boxSize = a_box.size();
1628 const RealVect x = this->position(a_index);
1629
1630 std::size_t lin = 0;
1631 std::size_t stride = 1;
1632 for (int d = 0; d < SpaceDim; d++) {
1633 int c = static_cast<int>(std::floor((x[d] - a_probLo[d]) / a_dx[d])) - loEnd[d];
1634 if (c < 0) {
1635 c = 0;
1636 }
1637 if (c >= boxSize[d]) {
1638 c = boxSize[d] - 1;
1639 }
1640 lin += static_cast<std::size_t>(c) * stride;
1641 stride *= static_cast<std::size_t>(boxSize[d]);
1642 }
1643
1644 return lin;
1645 }
