1611 void
1612 growTo(const std::size_t a_needed)
1613 {
1614 if (a_needed <= m_capacity) {
1615 return;
1616 }
1617
1618 // Saturating geometric growth: 2 * m_capacity overflows std::size_t once m_capacity exceeds
1619 // SIZE_MAX/2, and reserve()'s "a_capacity <= m_capacity" guard would then silently no-op.
1620 const std::size_t doubled = (m_capacity == 0) ? 16
1621 : (m_capacity <= std::numeric_limits<std::size_t>::max() / 2
1622 ? 2 * m_capacity
1623 : std::numeric_limits<std::size_t>::max());
1624
1625 this->reserve(std::max(a_needed, doubled));
1626 }
