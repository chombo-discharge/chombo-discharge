109{
110#ifdef CH_USE_MEMORY_TRACKING
111 std::size_t seen = a_mark.load(std::memory_order_relaxed);
112
113 while (a_value > seen && !a_mark.compare_exchange_weak(seen, a_value, std::memory_order_relaxed)) {
114 }
115#endif
116}
